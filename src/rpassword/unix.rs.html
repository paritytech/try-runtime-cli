14/// Turns a C function return into an IO Result
15fn io_result(ret: c_int) -> std::io::Result<()> {
16    match ret {
17        0 => Ok(()),
18        _ => Err(std::io::Error::last_os_error()),
19    }
20}

22fn is_interactive_terminal(fd: c_int) -> bool {
23    unsafe { isatty(fd) != 0 }
24}
25
26fn safe_tcgetattr(fd: c_int) -> std::io::Result<termios> {
27    let mut term = mem::MaybeUninit::<termios>::uninit();
28    io_result(unsafe { ::libc::tcgetattr(fd, term.as_mut_ptr()) })?;
29    Ok(unsafe { term.assume_init() })
30}
31
32fn safe_tcsetattr(fd: c_int, term: &mut termios) -> std::io::Result<()> {
33    io_result(unsafe { tcsetattr(fd, TCSANOW, term) })
34}

47impl Drop for RawModeInput {
48    fn drop(&mut self) {
49        if let (Some(fd), Some(term_orig)) = (self.input_fd, self.input_term_orig.as_mut()) {
50            unsafe {
51                tcsetattr(fd, TCSANOW, term_orig);
52            }
53        }
54        if let (Some(fd), Some(term_orig)) = (self.output_fd, self.output_term_orig.as_mut()) {
55            unsafe {
56                tcsetattr(fd, TCSANOW, term_orig);
57            }
58        }
59    }
60}

62impl RawPasswordInput for RawModeInput {
63    fn new(config: Config) -> io::Result<impl RawPasswordInput> {
64        let mut input_fd: Option<RawFd> = None;
65        let input: Box<dyn Read> = match config.input {
66            InputTarget::FilePath(path) => {
67                let file = OpenOptions::new().read(true).open(path)?;
68                input_fd = Some(file.as_raw_fd());
69                Box::new(file)
70            }
71            InputTarget::Reader(reader) => Box::new(reader),
72        };
73        let input_is_tty = if let Some(fd) = input_fd {
74            is_interactive_terminal(fd)
75        } else {
76            false
77        };
78        let input_term_orig = if input_is_tty {
79            if let Some(fd) = input_fd {
80                Some(safe_tcgetattr(fd)?)
81            } else {
82                None
83            }
84        } else {
85            None
86        };
87
88        let mut output_fd: Option<RawFd> = None;
89        let output: Box<dyn Write> = match config.output {
90            OutputTarget::FilePath(path) => {
91                let file = OpenOptions::new().write(true).open(path)?;
92                output_fd = Some(file.as_raw_fd());
93                Box::new(file)
94            }
95            OutputTarget::Writer(writer) => Box::new(writer),
96            OutputTarget::Void => Box::new(Cursor::new(Vec::<u8>::new())), // TODO: Should use a SafeVec instead
97        };
98        let output_is_tty = if let Some(fd) = output_fd {
99            is_interactive_terminal(fd)
100        } else {
101            false
102        };
103        let output_term_orig = if output_is_tty {
104            if let Some(fd) = output_fd {
105                Some(safe_tcgetattr(fd)?)
106            } else {
107                None
108            }
109        } else {
110            None
111        };
112
113        Ok(RawModeInput {
114            input,
115            input_fd,
116            input_term_orig,
117            input_is_tty,
118            output,
119            output_fd,
120            output_term_orig,
121            output_is_tty,
122        })
123    }
124
125    fn needs_terminal_configuration(&self) -> bool {
126        self.input_is_tty
127    }
128
129    fn apply_terminal_configuration(&mut self) -> io::Result<()> {
130        if self.input_is_tty {
131            if let Some(fd) = self.input_fd {
132                let mut term = safe_tcgetattr(fd)?;
133                term.c_lflag &= !(ECHO | ICANON | ECHONL | ISIG);
134                term.c_cc[VMIN] = 1;
135                term.c_cc[VTIME] = 0;
136                safe_tcsetattr(fd, &mut term)?;
137            }
138        }
139
140        if self.output_is_tty {
141            if let Some(fd) = self.output_fd {
142                let mut term = safe_tcgetattr(fd)?;
143                term.c_lflag &= !(ECHO | ICANON | ECHONL | ISIG);
144                term.c_cc[VMIN] = 1;
145                term.c_cc[VTIME] = 0;
146                safe_tcsetattr(fd, &mut term)?;
147            }
148        }
149
150        Ok(())
151    }
152
153    fn read_char(&mut self) -> std::io::Result<char> {
154        read_char(&mut self.input)
155    }
156
157    fn write_output(&mut self, output: &str) -> std::io::Result<()> {
158        self.output.write_all(output.as_bytes())?;
159        self.output.flush()
160    }
161
162    fn send_signal_sigint(&mut self) -> io::Result<()> {
163        if unsafe { libc::raise(libc::SIGINT) != 0 } {
164            return Err(std::io::Error::last_os_error());
165        }
166        Ok(())
167    }
168}
