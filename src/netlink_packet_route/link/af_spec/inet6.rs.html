22const IFLA_INET6_FLAGS: u16 = 1;
23const IFLA_INET6_CONF: u16 = 2;
24const IFLA_INET6_STATS: u16 = 3;
25// No kernel code used IFLA_INET6_MCAST
26// const IFLA_INET6_MCAST: u16 = 4;
27const IFLA_INET6_CACHEINFO: u16 = 5;
28const IFLA_INET6_ICMP6STATS: u16 = 6;
29const IFLA_INET6_TOKEN: u16 = 7;
30const IFLA_INET6_ADDR_GEN_MODE: u16 = 8;
31const IFLA_INET6_RA_MTU: u16 = 9;

33#[derive(Clone, Eq, PartialEq, Debug)]
34#[non_exhaustive]
35pub enum AfSpecInet6 {
36    //TODO(Gris Ge): Use Vec<enum> for `IFF_UP` and etc
37    Flags(Inet6IfaceFlags),
38    CacheInfo(Inet6CacheInfo),
39    DevConf(Inet6DevConf),
40    Stats(Inet6Stats),
41    Icmp6Stats(Icmp6Stats),
42    Token(Ipv6Addr),
43    AddrGenMode(In6AddrGenMode),
44    RaMtu(u32),
45    Other(DefaultNla),
46}
47
48pub(crate) struct VecAfSpecInet6(pub(crate) Vec<AfSpecInet6>);
49
50impl<'a, T: AsRef<[u8]> + ?Sized> Parseable<NlaBuffer<&'a T>>
51    for VecAfSpecInet6
52{
53    fn parse(buf: &NlaBuffer<&'a T>) -> Result<Self, DecodeError> {
54        let mut nlas = vec![];
55        let err = "Invalid AF_INET6 NLA for IFLA_AF_SPEC(AF_UNSPEC)";
56        for nla in NlasIterator::new(buf.into_inner()) {
57            let nla = nla.context(err)?;
58            nlas.push(AfSpecInet6::parse(&nla).context(err)?);
59        }
60        Ok(Self(nlas))
61    }
62}

64impl Nla for AfSpecInet6 {
65    fn value_len(&self) -> usize {
66        use self::AfSpecInet6::*;
67        match *self {
68            CacheInfo(ref cache_info) => cache_info.buffer_len(),
69            DevConf(ref dev_conf) => dev_conf.buffer_len(),
70            Stats(ref stats) => stats.buffer_len(),
71            Icmp6Stats(ref icmp_stats) => icmp_stats.buffer_len(),
72            Flags(_) | RaMtu(_) => 4,
73            Token(_) => 16,
74            AddrGenMode(_) => 1,
75            Other(ref nla) => nla.value_len(),
76        }
77    }
78
79    fn emit_value(&self, buffer: &mut [u8]) {
80        use self::AfSpecInet6::*;
81        match *self {
82            Flags(ref value) => emit_u32(buffer, value.bits()).unwrap(),
83            RaMtu(ref value) => emit_u32(buffer, *value).unwrap(),
84            CacheInfo(ref v) => v.emit(buffer),
85            DevConf(ref v) => v.emit(buffer),
86            Stats(ref v) => v.emit(buffer),
87            Icmp6Stats(ref v) => v.emit(buffer),
88            Token(v) => buffer.copy_from_slice(&v.octets()),
89            AddrGenMode(ref v) => buffer[0] = v.into(),
90            Other(ref nla) => nla.emit_value(buffer),
91        }
92    }
93
94    fn kind(&self) -> u16 {
95        use self::AfSpecInet6::*;
96        match *self {
97            Flags(_) => IFLA_INET6_FLAGS,
98            CacheInfo(_) => IFLA_INET6_CACHEINFO,
99            DevConf(_) => IFLA_INET6_CONF,
100            Stats(_) => IFLA_INET6_STATS,
101            Icmp6Stats(_) => IFLA_INET6_ICMP6STATS,
102            Token(_) => IFLA_INET6_TOKEN,
103            AddrGenMode(_) => IFLA_INET6_ADDR_GEN_MODE,
104            RaMtu(_) => IFLA_INET6_RA_MTU,
105            Other(ref nla) => nla.kind(),
106        }
107    }
108}

110impl<'a, T: AsRef<[u8]> + ?Sized> Parseable<NlaBuffer<&'a T>> for AfSpecInet6 {
111    fn parse(buf: &NlaBuffer<&'a T>) -> Result<Self, DecodeError> {
112        use self::AfSpecInet6::*;
113
114        let payload = buf.value();
115        Ok(match buf.kind() {
116            IFLA_INET6_FLAGS => Flags(Inet6IfaceFlags::from_bits_retain(
117                parse_u32(payload).context("invalid IFLA_INET6_FLAGS value")?,
118            )),
119            IFLA_INET6_CACHEINFO => CacheInfo(
120                Inet6CacheInfo::parse(&Inet6CacheInfoBuffer::new(payload))
121                    .context(format!(
122                        "invalid IFLA_INET6_CACHEINFO value {payload:?}"
123                    ))?,
124            ),
125            IFLA_INET6_CONF => DevConf(
126                Inet6DevConf::parse(&Inet6DevConfBuffer::new(
127                    expand_buffer_if_small(
128                        payload,
129                        LINK_INET6_DEV_CONF_LEN,
130                        "IFLA_INET6_CONF",
131                    )
132                    .as_slice(),
133                ))
134                .context(format!(
135                    "invalid IFLA_INET6_CONF value {payload:?}"
136                ))?,
137            ),
138            IFLA_INET6_STATS => Stats(
139                Inet6Stats::parse(&Inet6StatsBuffer::new(
140                    expand_buffer_if_small(
141                        payload,
142                        INET6_STATS_LEN,
143                        "IFLA_INET6_STATS",
144                    )
145                    .as_slice(),
146                ))
147                .context(format!(
148                    "invalid IFLA_INET6_STATS value {payload:?}"
149                ))?,
150            ),
151            IFLA_INET6_ICMP6STATS => Icmp6Stats(
152                super::super::Icmp6Stats::parse(&Icmp6StatsBuffer::new(
153                    expand_buffer_if_small(
154                        payload,
155                        ICMP6_STATS_LEN,
156                        "IFLA_INET6_ICMP6STATS",
157                    )
158                    .as_slice(),
159                ))
160                .context(format!(
161                    "invalid IFLA_INET6_ICMP6STATS value {payload:?}"
162                ))?,
163            ),
164            IFLA_INET6_TOKEN => Token(
165                parse_ipv6_addr(payload)
166                    .context("invalid IFLA_INET6_TOKEN value")?,
167            ),
168            IFLA_INET6_ADDR_GEN_MODE => {
169                let mode = parse_u8(payload)
170                    .context("invalid IFLA_INET6_ADDR_GEN_MODE")?;
171                AddrGenMode(In6AddrGenMode::from(mode))
172            }
173            IFLA_INET6_RA_MTU => RaMtu(
174                parse_u32(payload)
175                    .context("invalid IFLA_INET6_RA_MTU value")?,
176            ),
177            kind => Other(DefaultNla::parse(buf).context(format!(
178                "unknown AF_INET6 NLA type {kind} for IFLA_AF_SPEC(AF_UNSPEC)"
179            ))?),
180        })
181    }
