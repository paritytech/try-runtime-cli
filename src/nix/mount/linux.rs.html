93/// Mount a file system.

95/// # Arguments

97/// - `target` -    Specifies the destination.  e.g. `/mnt`.

99/// - `flags` -     Optional flags controlling the mount.
100/// - `data` -      Optional file system specific data.

135/// Unmount the file system mounted at `target`.
136pub fn umount<P: ?Sized + NixPath>(target: &P) -> Result<()> {
137    let res =
138        target.with_nix_path(|cstr| unsafe { libc::umount(cstr.as_ptr()) })?;
139
140    Errno::result(res).map(drop)
141}
