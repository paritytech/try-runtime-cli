1//! Wasm compilation orchestration.
2//!
3//! It works roughly like this:
4//!
5//! * We walk over the Wasm module/component and make a list of all the things
6//!   we need to compile. This is a `CompileInputs`.
7//!
8//! * The `CompileInputs::compile` method compiles each of these in parallel,
9//!   producing a `UnlinkedCompileOutputs`. This is an unlinked set of compiled
10//!   functions, bucketed by type of function.
11//!
12//! * The `UnlinkedCompileOutputs::pre_link` method re-arranges the compiled
13//!   functions into a flat list. This is the order we will place them within
14//!   the ELF file, so we must also keep track of all the functions' indices
15//!   within this list, because we will need them for resolving
16//!   relocations. These indices are kept track of in the resulting
17//!   `FunctionIndices`.
18//!
19//! * The `FunctionIndices::link_and_append_code` method appends the functions
20//!   to the given ELF file and resolves relocations. It produces an `Artifacts`
21//!   which contains the data needed at runtime to find and call Wasm
22//!   functions. It is up to the caller to serialize the relevant parts of the
23//!   `Artifacts` into the ELF file.
24

62/// Converts an input binary-encoded WebAssembly module to compilation
63/// artifacts and type information.
64///
65/// This is where compilation actually happens of WebAssembly modules and
66/// translation/parsing/validation of the binary input occurs. The binary
67/// artifact represented in the `MmapVec` returned here is an in-memory ELF
68/// file in an owned area of virtual linear memory where permissions (such
69/// as the executable bit) can be applied.
70///
71/// Additionally compilation returns an `Option` here which is always
72/// `Some`, notably compiled metadata about the module in addition to the
73/// type information found within.
74pub(crate) fn build_artifacts<T: FinishedObject>(
75    engine: &Engine,
76    wasm: &[u8],
77    dwarf_package: Option<&[u8]>,
78    obj_state: &T::State,
79) -> Result<(T, Option<(CompiledModuleInfo, ModuleTypes)>)> {
80    let tunables = engine.tunables();
81
82    // First a `ModuleEnvironment` is created which records type information
83    // about the wasm module. This is where the WebAssembly is parsed and
84    // validated. Afterwards `types` will have all the type information for
85    // this module.
86    let mut parser = wasmparser::Parser::new(0);
87    let mut validator = wasmparser::Validator::new_with_features(engine.features());
88    parser.set_features(*validator.features());
89    let mut types = ModuleTypesBuilder::new(&validator);
90    let mut translation = ModuleEnvironment::new(tunables, &mut validator, &mut types)
91        .translate(parser, wasm)
92        .context("failed to parse WebAssembly module")?;
93    let functions = mem::take(&mut translation.function_body_inputs);
94
95    let compile_inputs = CompileInputs::for_module(&types, &translation, functions);
96    let unlinked_compile_outputs = compile_inputs.compile(engine)?;
97    let PreLinkOutput {
98        needs_gc_heap,
99        compiled_funcs,
100        indices,
101    } = unlinked_compile_outputs.pre_link();
102    translation.module.needs_gc_heap |= needs_gc_heap;
103
104    // Emplace all compiled functions into the object file with any other
105    // sections associated with code as well.
106    let mut object = engine.compiler().object(ObjectKind::Module)?;
107    // Insert `Engine` and type-level information into the compiled
108    // artifact so if this module is deserialized later it contains all
109    // information necessary.
110    //
111    // Note that `append_compiler_info` and `append_types` here in theory
112    // can both be skipped if this module will never get serialized.
113    // They're only used during deserialization and not during runtime for
114    // the module itself. Currently there's no need for that, however, so
115    // it's left as an exercise for later.
116    engine.append_compiler_info(&mut object);
117    engine.append_bti(&mut object);
118
119    let (mut object, compilation_artifacts) = indices.link_and_append_code(
120        &types,
121        object,
122        engine,
123        compiled_funcs,
124        std::iter::once(translation).collect(),
125        dwarf_package,
126    )?;
127
128    let info = compilation_artifacts.unwrap_as_module_info();
129    let types = types.finish();
130    object.serialize_info(&(&info, &types));
131    let result = T::finish_object(object, obj_state)?;
132
133    Ok((result, Some((info, types))))
134}

136/// Performs the compilation phase for a component, translating and
137/// validating the provided wasm binary to machine code.
138///
139/// This method will compile all nested core wasm binaries in addition to
140/// any necessary extra functions required for operation with components.
141/// The output artifact here is the serialized object file contained within
142/// an owned mmap along with metadata about the compilation itself.
143#[cfg(feature = "component-model")]
144pub(crate) fn build_component_artifacts<T: FinishedObject>(
145    engine: &Engine,
146    binary: &[u8],
147    _dwarf_package: Option<&[u8]>,
148    obj_state: &T::State,
149) -> Result<(T, Option<wasmtime_environ::component::ComponentArtifacts>)> {
150    use wasmtime_environ::ScopeVec;
151    use wasmtime_environ::component::{
152        CompiledComponentInfo, ComponentArtifacts, ComponentTypesBuilder,
153    };
154
155    let tunables = engine.tunables();
156    let compiler = engine.compiler();
157
158    let scope = ScopeVec::new();
159    let mut validator = wasmparser::Validator::new_with_features(engine.features());
160    let mut types = ComponentTypesBuilder::new(&validator);
161    let (component, mut module_translations) =
162        Translator::new(tunables, &mut validator, &mut types, &scope)
163            .translate(binary)
164            .context("failed to parse WebAssembly module")?;
165
166    let compile_inputs = CompileInputs::for_component(
167        engine,
168        &types,
169        &component,
170        module_translations.iter_mut().map(|(i, translation)| {
171            let functions = mem::take(&mut translation.function_body_inputs);
172            (i, &*translation, functions)
173        }),
174    );
175    let unlinked_compile_outputs = compile_inputs.compile(&engine)?;
176
177    let PreLinkOutput {
178        needs_gc_heap,
179        compiled_funcs,
180        indices,
181    } = unlinked_compile_outputs.pre_link();
182    for (_, t) in &mut module_translations {
183        t.module.needs_gc_heap |= needs_gc_heap
184    }
185
186    let mut object = compiler.object(ObjectKind::Component)?;
187    engine.append_compiler_info(&mut object);
188    engine.append_bti(&mut object);
189
190    let (mut object, compilation_artifacts) = indices.link_and_append_code(
191        types.module_types_builder(),
192        object,
193        engine,
194        compiled_funcs,
195        module_translations,
196        None, // TODO: Support dwarf packages for components.
197    )?;
198    let (types, ty) = types.finish(&component.component);
199
200    let info = CompiledComponentInfo {
201        component: component.component,
202        trampolines: compilation_artifacts.trampolines,
203        resource_drop_wasm_to_array_trampoline: compilation_artifacts
204            .resource_drop_wasm_to_array_trampoline,
205    };
206    let artifacts = ComponentArtifacts {
207        info,
208        ty,
209        types,
210        static_modules: compilation_artifacts.modules,
211    };
212    object.serialize_info(&artifacts);
213
214    let result = T::finish_object(object, obj_state)?;
215    Ok((result, Some(artifacts)))
216}

220/// A sortable, comparable key for a compilation output.
221///
222/// Two `u32`s to align with `cranelift_codegen::ir::UserExternalName`.
223#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
224struct CompileKey {
225    // The namespace field is bitpacked like:
226    //
227    //     [ kind:i3 module:i29 ]
228    namespace: u32,
229
230    index: u32,
231}
232
233#[repr(u32)]
234#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
235enum CompileKind {
236    WasmFunction = CompileKey::new_kind(0),
237    ArrayToWasmTrampoline = CompileKey::new_kind(1),
238    WasmToArrayTrampoline = CompileKey::new_kind(2),
239    WasmToBuiltinTrampoline = CompileKey::new_kind(3),
240
241    #[cfg(feature = "component-model")]
242    Trampoline = CompileKey::new_kind(4),
243    #[cfg(feature = "component-model")]
244    ResourceDropWasmToArrayTrampoline = CompileKey::new_kind(5),
245}
246
247impl From<CompileKind> for u32 {
248    fn from(kind: CompileKind) -> Self {
249        kind as u32
250    }
251}
252
253impl core::fmt::Debug for CompileKey {
254    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
255        f.debug_struct("CompileKey")
256            .field("kind", &self.kind())
257            .field("module", &self.module())
258            .field("index", &self.index)
259            .finish()
260    }
261}

263impl CompileKey {
264    const KIND_BITS: u32 = 3;
265    const KIND_OFFSET: u32 = 32 - Self::KIND_BITS;
266    const KIND_MASK: u32 = ((1 << Self::KIND_BITS) - 1) << Self::KIND_OFFSET;
267
268    const fn new_kind(kind: u32) -> u32 {
269        assert!(kind < (1 << Self::KIND_BITS));
270        kind << Self::KIND_OFFSET
271    }
272
273    fn kind(&self) -> CompileKind {
274        let k = self.namespace & Self::KIND_MASK;
275        if k == u32::from(CompileKind::WasmFunction) {
276            return CompileKind::WasmFunction;
277        }
278        if k == u32::from(CompileKind::ArrayToWasmTrampoline) {
279            return CompileKind::ArrayToWasmTrampoline;
280        }
281        if k == u32::from(CompileKind::WasmToArrayTrampoline) {
282            return CompileKind::WasmToArrayTrampoline;
283        }
284        if k == u32::from(CompileKind::WasmToBuiltinTrampoline) {
285            return CompileKind::WasmToBuiltinTrampoline;
286        }
287
288        #[cfg(feature = "component-model")]
289        {
290            if k == u32::from(CompileKind::Trampoline) {
291                return CompileKind::Trampoline;
292            }
293            if k == u32::from(CompileKind::ResourceDropWasmToArrayTrampoline) {
294                return CompileKind::ResourceDropWasmToArrayTrampoline;
295            }
296        }
297
298        unreachable!()
299    }
300
301    fn module(&self) -> StaticModuleIndex {
302        StaticModuleIndex::from_u32(self.namespace & !Self::KIND_MASK)
303    }
304
305    fn defined_func_index(&self) -> DefinedFuncIndex {
306        DefinedFuncIndex::from_u32(self.index)
307    }
308
309    // NB: more kinds in the other `impl` block.
310
311    fn wasm_function(module: StaticModuleIndex, index: DefinedFuncIndex) -> Self {
312        debug_assert_eq!(module.as_u32() & Self::KIND_MASK, 0);
313        Self {
314            namespace: u32::from(CompileKind::WasmFunction) | module.as_u32(),
315            index: index.as_u32(),
316        }
317    }
318
319    fn array_to_wasm_trampoline(module: StaticModuleIndex, index: DefinedFuncIndex) -> Self {
320        debug_assert_eq!(module.as_u32() & Self::KIND_MASK, 0);
321        Self {
322            namespace: u32::from(CompileKind::ArrayToWasmTrampoline) | module.as_u32(),
323            index: index.as_u32(),
324        }
325    }
326
327    fn wasm_to_array_trampoline(index: ModuleInternedTypeIndex) -> Self {
328        Self {
329            namespace: CompileKind::WasmToArrayTrampoline.into(),
330            index: index.as_u32(),
331        }
332    }
333
334    fn wasm_to_builtin_trampoline(index: BuiltinFunctionIndex) -> Self {
335        Self {
336            namespace: CompileKind::WasmToBuiltinTrampoline.into(),
337            index: index.index(),
338        }
339    }
340}

359#[derive(Clone, Copy)]
360enum CompiledFunction<T> {
361    Function(T),
362    #[cfg(feature = "component-model")]
363    AllCallFunc(wasmtime_environ::component::AllCallFunc<T>),
364}
365
366impl<T> CompiledFunction<T> {
367    fn as_function(&self) -> Option<&T> {
368        match self {
369            Self::Function(f) => Some(f),
370            #[cfg(feature = "component-model")]
371            Self::AllCallFunc(_) => None,
372        }
373    }
374
375    fn as_function_mut(&mut self) -> Option<&mut T> {
376        match self {
377            Self::Function(f) => Some(f),
378            #[cfg(feature = "component-model")]
379            Self::AllCallFunc(_) => None,
380        }
381    }
382
383    fn unwrap_function(self) -> T {
384        match self {
385            Self::Function(f) => f,
386            #[cfg(feature = "component-model")]
387            Self::AllCallFunc(_) => panic!("CompiledFunction::unwrap_function"),
388        }
389    }
390
391    #[cfg(feature = "component-model")]
392    fn unwrap_all_call_func(self) -> wasmtime_environ::component::AllCallFunc<T> {
393        match self {
394            Self::AllCallFunc(f) => f,
395            Self::Function(_) => panic!("CompiledFunction::unwrap_all_call_func"),
396        }
397    }
398}

429/// The collection of things we need to compile for a Wasm module or component.
430#[derive(Default)]
431struct CompileInputs<'a> {
432    inputs: Vec<CompileInput<'a>>,
433}
434
435impl<'a> CompileInputs<'a> {
436    fn push_input(
437        &mut self,
438        f: impl FnOnce(&dyn Compiler) -> Result<CompileOutput<'a>> + Send + 'a,
439    ) {
440        self.inputs.push(Box::new(f));
441    }
442
443    /// Create the `CompileInputs` for a core Wasm module.
444    fn for_module(
445        types: &'a ModuleTypesBuilder,
446        translation: &'a ModuleTranslation<'a>,
447        functions: PrimaryMap<DefinedFuncIndex, FunctionBodyData<'a>>,
448    ) -> Self {
449        let mut ret = CompileInputs { inputs: vec![] };
450
451        let module_index = StaticModuleIndex::from_u32(0);
452        ret.collect_inputs_in_translations(types, [(module_index, translation, functions)]);
453
454        ret
455    }
456
457    /// Create a `CompileInputs` for a component.
458    #[cfg(feature = "component-model")]
459    fn for_component(
460        engine: &'a Engine,
461        types: &'a wasmtime_environ::component::ComponentTypesBuilder,
462        component: &'a wasmtime_environ::component::ComponentTranslation,
463        module_translations: impl IntoIterator<
464            Item = (
465                StaticModuleIndex,
466                &'a ModuleTranslation<'a>,
467                PrimaryMap<DefinedFuncIndex, FunctionBodyData<'a>>,
468            ),
469        >,
470    ) -> Self {
471        let mut ret = CompileInputs { inputs: vec![] };
472
473        ret.collect_inputs_in_translations(types.module_types_builder(), module_translations);
474        let tunables = engine.tunables();
475
476        for (idx, trampoline) in component.trampolines.iter() {
477            ret.push_input(move |compiler| {
478                let symbol = trampoline.symbol_name();
479                Ok(CompileOutput {
480                    key: CompileKey::trampoline(idx),
481                    function: compiler
482                        .component_compiler()
483                        .compile_trampoline(component, types, idx, tunables, &symbol)
484                        .with_context(|| format!("failed to compile {symbol}"))?
485                        .into(),
486                    symbol,
487                    start_srcloc: FilePos::default(),
488                    translation: None,
489                    func_body: None,
490                })
491            });
492        }
493
494        // If there are any resources defined within this component, the
495        // signature for `resource.drop` is mentioned somewhere, and the
496        // wasm-to-native trampoline for `resource.drop` hasn't been created yet
497        // then insert that here. This is possibly required by destruction of
498        // resources from the embedder and otherwise won't be explicitly
499        // requested through initializers above or such.
500        if component.component.num_resources > 0 {
501            if let Some(sig) = types.find_resource_drop_signature() {
502                ret.push_input(move |compiler| {
503                    let symbol = "resource_drop_trampoline".to_string();
504                    let function = compiler
505                        .compile_wasm_to_array_trampoline(types[sig].unwrap_func(), &symbol)
506                        .with_context(|| format!("failed to compile `{symbol}`"))?;
507                    Ok(CompileOutput {
508                        key: CompileKey::resource_drop_wasm_to_array_trampoline(),
509                        function: CompiledFunction::Function(function),
510                        symbol,
511                        start_srcloc: FilePos::default(),
512                        translation: None,
513                        func_body: None,
514                    })
515                });
516            }
517        }
518
519        ret
520    }

522    fn clean_symbol(name: &str) -> Cow<'_, str> {
523        /// Maximum length of symbols generated in objects.
524        const MAX_SYMBOL_LEN: usize = 96;
525
526        // Just to be on the safe side, filter out characters that could
527        // pose issues to tools such as "perf" or "objdump".  To avoid
528        // having to update a list of allowed characters for each different
529        // language that compiles to Wasm, allows only graphic ASCII
530        // characters; replace runs of everything else with a "?".
531        let bad_char = |c: char| !c.is_ascii_graphic();
532        if name.chars().any(bad_char) {
533            let mut last_char_seen = '\u{0000}';
534            Cow::Owned(
535                name.chars()
536                    .map(|c| if bad_char(c) { '?' } else { c })
537                    .filter(|c| {
538                        let skip = last_char_seen == '?' && *c == '?';
539                        last_char_seen = *c;
540                        !skip
541                    })
542                    .take(MAX_SYMBOL_LEN)
543                    .collect::<String>(),
544            )
545        } else if name.len() <= MAX_SYMBOL_LEN {
546            Cow::Borrowed(&name[..])
547        } else {
548            Cow::Borrowed(&name[..MAX_SYMBOL_LEN])
549        }
550    }
551
552    fn collect_inputs_in_translations(
553        &mut self,
554        types: &'a ModuleTypesBuilder,
555        translations: impl IntoIterator<
556            Item = (
557                StaticModuleIndex,
558                &'a ModuleTranslation<'a>,
559                PrimaryMap<DefinedFuncIndex, FunctionBodyData<'a>>,
560            ),
561        >,
562    ) {
563        for (module, translation, functions) in translations {
564            for (def_func_index, func_body_data) in functions {
565                self.push_input(move |compiler| {
566                    let func_index = translation.module.func_index(def_func_index);
567                    let symbol = match translation
568                        .debuginfo
569                        .name_section
570                        .func_names
571                        .get(&func_index)
572                    {
573                        Some(name) => format!(
574                            "wasm[{}]::function[{}]::{}",
575                            module.as_u32(),
576                            func_index.as_u32(),
577                            Self::clean_symbol(&name)
578                        ),
579                        None => format!(
580                            "wasm[{}]::function[{}]",
581                            module.as_u32(),
582                            func_index.as_u32()
583                        ),
584                    };
585                    let func_body = func_body_data.body.clone();
586                    let data = func_body.get_binary_reader();
587                    let offset = data.original_position();
588                    let start_srcloc = FilePos::new(u32::try_from(offset).unwrap());
589                    let function = compiler
590                        .compile_function(
591                            translation,
592                            def_func_index,
593                            func_body_data,
594                            types,
595                            &symbol,
596                        )
597                        .with_context(|| format!("failed to compile: {symbol}"))?;
598
599                    Ok(CompileOutput {
600                        key: CompileKey::wasm_function(module, def_func_index),
601                        symbol,
602                        function: CompiledFunction::Function(function),
603                        start_srcloc,
604                        translation: Some(translation),
605                        func_body: Some(func_body),
606                    })
607                });
608
609                let func_index = translation.module.func_index(def_func_index);
610                if translation.module.functions[func_index].is_escaping() {
611                    self.push_input(move |compiler| {
612                        let func_index = translation.module.func_index(def_func_index);
613                        let symbol = format!(
614                            "wasm[{}]::array_to_wasm_trampoline[{}]",
615                            module.as_u32(),
616                            func_index.as_u32()
617                        );
618                        let trampoline = compiler
619                            .compile_array_to_wasm_trampoline(
620                                translation,
621                                types,
622                                def_func_index,
623                                &symbol,
624                            )
625                            .with_context(|| format!("failed to compile: {symbol}"))?;
626                        Ok(CompileOutput {
627                            key: CompileKey::array_to_wasm_trampoline(module, def_func_index),
628                            symbol,
629                            function: CompiledFunction::Function(trampoline),
630                            start_srcloc: FilePos::default(),
631                            translation: None,
632                            func_body: None,
633                        })
634                    });
635                }
636            }
637        }
638
639        let mut trampoline_types_seen = HashSet::new();
640        for (_func_type_index, trampoline_type_index) in types.trampoline_types() {
641            let is_new = trampoline_types_seen.insert(trampoline_type_index);
642            if !is_new {
643                continue;
644            }
645            let trampoline_func_ty = types[trampoline_type_index].unwrap_func();
646            self.push_input(move |compiler| {
647                let symbol = format!(
648                    "signatures[{}]::wasm_to_array_trampoline",
649                    trampoline_type_index.as_u32()
650                );
651                let trampoline = compiler
652                    .compile_wasm_to_array_trampoline(trampoline_func_ty, &symbol)
653                    .with_context(|| format!("failed to compile: {symbol}"))?;
654                Ok(CompileOutput {
655                    key: CompileKey::wasm_to_array_trampoline(trampoline_type_index),
656                    function: CompiledFunction::Function(trampoline),
657                    symbol,
658                    start_srcloc: FilePos::default(),
659                    translation: None,
660                    func_body: None,
661                })
662            });
663        }
664    }

666    /// Compile these `CompileInput`s (maybe in parallel) and return the
667    /// resulting `UnlinkedCompileOutput`s.
668    fn compile(self, engine: &Engine) -> Result<UnlinkedCompileOutputs<'a>> {
669        let compiler = engine.compiler();
670
671        if self.inputs.len() > 0 && cfg!(miri) {
672            bail!(
673                "\
674You are attempting to compile a WebAssembly module or component that contains
675functions in Miri. Running Cranelift through Miri is known to take quite a long
676time and isn't what we want in CI at least. If this is a mistake then you should
677ignore this test in Miri with:
678
679    #[cfg_attr(miri, ignore)]
680
681If this is not a mistake then try to edit the `pulley_provenance_test` test
682which runs Cranelift outside of Miri. If you still feel this is a mistake then
683please open an issue or a topic on Zulip to talk about how best to accomodate
684the use case.
685"
686            );
687        }
688
689        let mut raw_outputs = if let Some(inlining_compiler) = compiler.inlining_compiler() {
690            if engine.tunables().inlining {
691                self.compile_with_inlining(engine, compiler, inlining_compiler)?
692            } else {
693                // Inlining compiler but inlining is disabled: compile each
694                // input and immediately finish its output in parallel, skipping
695                // call graph computation and all that.
696                engine.run_maybe_parallel::<_, _, Error, _>(self.inputs, |f| {
697                    let mut compiled = f(compiler)?;
698                    match &mut compiled.function {
699                        CompiledFunction::Function(f) => inlining_compiler.finish_compiling(
700                            f,
701                            compiled.func_body.take(),
702                            &compiled.symbol,
703                        )?,
704                        #[cfg(feature = "component-model")]
705                        CompiledFunction::AllCallFunc(f) => {
706                            debug_assert!(compiled.func_body.is_none());
707                            inlining_compiler.finish_compiling(
708                                &mut f.array_call,
709                                None,
710                                &compiled.symbol,
711                            )?;
712                            inlining_compiler.finish_compiling(
713                                &mut f.wasm_call,
714                                None,
715                                &compiled.symbol,
716                            )?;
717                        }
718                    };
719                    Ok(compiled)
720                })?
721            }
722        } else {
723            // No inlining: just compile each individual input in parallel.
724            engine.run_maybe_parallel(self.inputs, |f| f(compiler))?
725        };
726
727        // Now that all functions have been compiled see if any
728        // wasmtime-builtin functions are necessary. If so those need to be
729        // collected and then those trampolines additionally need to be
730        // compiled.
731        compile_required_builtins(engine, &mut raw_outputs)?;
732
733        // Bucket the outputs by kind.
734        let mut outputs: BTreeMap<CompileKind, Vec<CompileOutput>> = BTreeMap::new();
735        for output in raw_outputs {
736            outputs.entry(output.key.kind()).or_default().push(output);
737        }
738
739        Ok(UnlinkedCompileOutputs { outputs })
740    }
741
742    fn compile_with_inlining(
743        self,
744        engine: &Engine,
745        compiler: &dyn Compiler,
746        inlining_compiler: &dyn InliningCompiler,
747    ) -> Result<Vec<CompileOutput<'a>>, Error> {
748        /// The index of a function (of any kind: Wasm function, trampoline, or
749        /// etc...) in our list of unlinked outputs.
750        #[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
751        struct OutputIndex(u32);
752        wasmtime_environ::entity_impl!(OutputIndex);
753
754        // Our list of unlinked outputs.
755        let mut outputs = PrimaryMap::<OutputIndex, Option<CompileOutput<'_>>>::from(
756            engine.run_maybe_parallel(self.inputs, |f| f(compiler).map(Some))?,
757        );
758
759        /// Get just the output indices of the Wasm functions from our unlinked
760        /// outputs.
761        fn wasm_functions<'a>(
762            outputs: &'a PrimaryMap<OutputIndex, Option<CompileOutput<'_>>>,
763        ) -> impl Iterator<Item = OutputIndex> + 'a {
764            outputs.iter().filter_map(|(i, o)| {
765                if o.as_ref()?.key.kind() == CompileKind::WasmFunction {
766                    Some(i)
767                } else {
768                    None
769                }
770            })
771        }
772
773        // A map from a Wasm function's (module, defined-function-index) pair to
774        // its index in our unlinked outputs.
775        //
776        // We will generally just be working with `OutputIndex`es, but
777        // occassionally we must translate from these pairs back to our index
778        // space, for example when we know that one module's function import is
779        // always satisfied with a particular function defined in a particular
780        // module. This map enables that translation.
781        let pair_to_output: HashMap<(StaticModuleIndex, DefinedFuncIndex), OutputIndex> = outputs
782            .iter()
783            .filter(|(_, output)| output.as_ref().unwrap().key.kind() == CompileKind::WasmFunction)
784            .map(|(output_index, output)| {
785                let output = output.as_ref().unwrap();
786                let module_index = output.key.module();
787                let defined_func_index = output.key.defined_func_index();
788                ((module_index, defined_func_index), output_index)
789            })
790            .collect();
791
792        // Construct the call graph for inlining.
793        //
794        // We only inline Wasm functions, not trampolines, because we rely on
795        // trampolines being in their own stack frame when we save the entry and
796        // exit SP, FP, and PC for backtraces in trampolines.
797        let call_graph = CallGraph::<OutputIndex>::new(wasm_functions(&outputs), {
798            let mut func_indices = IndexSet::default();
799            let outputs = &outputs;
800            let pair_to_output = &pair_to_output;
801            move |output_index, calls| {
802                debug_assert!(calls.is_empty());
803
804                let output = outputs[output_index].as_ref().unwrap();
805                debug_assert_eq!(output.key.kind(), CompileKind::WasmFunction);
806
807                let func = match &output.function {
808                    CompiledFunction::Function(f) => f,
809                    #[cfg(feature = "component-model")]
810                    CompiledFunction::AllCallFunc(_) => {
811                        unreachable!("wasm functions are not all-call functions")
812                    }
813                };
814
815                // Get this function's call graph edges as `FuncIndex`es.
816                func_indices.clear();
817                inlining_compiler.calls(func, &mut func_indices)?;
818
819                // Translate each of those to (module, defined-function-index)
820                // pairs and then finally to output indices, which is what we
821                // actually need.
822                let caller_module = output.key.module();
823                let translation = output
824                    .translation
825                    .expect("all wasm functions have translations");
826                calls.extend(func_indices.iter().copied().filter_map(|callee_func| {
827                    if let Some(callee_def_func) =
828                        translation.module.defined_func_index(callee_func)
829                    {
830                        // Call to a function in the same module.
831                        Some(pair_to_output[&(caller_module, callee_def_func)])
832                    } else if let Some(pair) = translation.known_imported_functions[callee_func] {
833                        // Call to a statically-known imported function.
834                        Some(pair_to_output[&pair])
835                    } else {
836                        // Call to an unknown imported function or perhaps to
837                        // multiple different functions in different
838                        // instantiations. Can't inline these calls, so don't
839                        // add them to the call graph.
840                        None
841                    }
842                }));
843                Ok(())
844            }
845        })?;
846
847        // Stratify the call graph into a sequence of layers. We process each
848        // layer in order, but process functions within a layer in parallel
849        // (because they either do not call each other or are part of a
850        // mutual-recursion cycle; either way we won't inline members of the
851        // same layer into each other).
852        let strata = stratify::Strata::<OutputIndex>::new(wasm_functions(&outputs), &call_graph);
853        let mut layer_outputs = vec![];
854        for layer in strata.layers() {
855            // Temporarily take this layer's outputs out of our unlinked outputs
856            // list so that we can mutate these outputs (by inlining callee
857            // functions into them) while also accessing shared borrows of the
858            // unlinked outputs list (finding the callee functions we will
859            // inline).
860            debug_assert!(layer_outputs.is_empty());
861            layer_outputs.extend(layer.iter().map(|f| outputs[*f].take().unwrap()));
862
863            // Process this layer's members in parallel.
864            engine.run_maybe_parallel_mut(
865                &mut layer_outputs,
866                |output: &mut CompileOutput<'_>| {
867                    debug_assert_eq!(output.key.kind(), CompileKind::WasmFunction);
868
869                    let caller_translation = output.translation.unwrap();
870                    let caller_module = output.key.module();
871                    let caller_def_func = output.key.defined_func_index();
872                    let caller_needs_gc_heap = caller_translation.module.needs_gc_heap;
873
874                    let caller = output
875                        .function
876                        .as_function_mut()
877                        .expect("wasm functions are not all-call functions");
878
879                    let mut caller_size = inlining_compiler.size(caller);
880
881                    inlining_compiler.inline(caller, &mut |callee: FuncIndex| {
882                        let (callee_module, callee_def_func, callee_needs_gc_heap) =
883                            if let Some(def_func) =
884                                caller_translation.module.defined_func_index(callee)
885                            {
886                                (caller_module, def_func, Some(caller_needs_gc_heap))
887                            } else {
888                                let (def_module, def_func) =
889                                    caller_translation.known_imported_functions[callee].expect(
890                                        "a direct call to an imported function must have a \
891                                         statically-known import",
892                                    );
893                                (def_module, def_func, None)
894                            };
895
896                        let callee_output_index: OutputIndex =
897                            pair_to_output[&(callee_module, callee_def_func)];
898                        let callee_output = outputs[callee_output_index].as_ref()?;
899                        let callee_needs_gc_heap = callee_needs_gc_heap.unwrap_or_else(|| {
900                            callee_output.translation.unwrap().module.needs_gc_heap
901                        });
902
903                        debug_assert_eq!(callee_output.key.kind(), CompileKind::WasmFunction);
904                        let callee = callee_output
905                            .function
906                            .as_function()
907                            .expect("wasm functions are not all-call functions");
908
909                        let callee_size = inlining_compiler.size(callee);
910
911                        if Self::should_inline(InlineHeuristicParams {
912                            tunables: engine.tunables(),
913                            caller_size,
914                            caller_module,
915                            caller_def_func,
916                            caller_needs_gc_heap,
917                            callee_size,
918                            callee_module,
919                            callee_def_func,
920                            callee_needs_gc_heap,
921                        }) {
922                            caller_size = caller_size.saturating_add(callee_size);
923                            Some(callee)
924                        } else {
925                            None
926                        }
927                    })
928                },
929            )?;
930
931            for (f, func) in layer.iter().zip(layer_outputs.drain(..)) {
932                debug_assert!(outputs[*f].is_none());
933                outputs[*f] = Some(func);
934            }
935        }
936
937        // Fan out in parallel again and finish compiling each function.
938        engine.run_maybe_parallel(outputs.into(), |output| {
939            let mut output = output.unwrap();
940            match &mut output.function {
941                CompiledFunction::Function(f) => inlining_compiler.finish_compiling(
942                    f,
943                    output.func_body.take(),
944                    &output.symbol,
945                )?,
946                #[cfg(feature = "component-model")]
947                CompiledFunction::AllCallFunc(f) => {
948                    debug_assert!(output.func_body.is_none());
949                    inlining_compiler.finish_compiling(&mut f.array_call, None, &output.symbol)?;
950                    inlining_compiler.finish_compiling(&mut f.wasm_call, None, &output.symbol)?;
951                }
952            };
953            Ok(output)
954        })
955    }
956
957    /// Implementation of our inlining heuristics.
958    ///
959    /// TODO: We should improve our heuristics:
960    ///
961    /// * One potentially promising hint that we don't currently make use of is
962    ///   how many times a function appears as the callee in call sites. For
963    ///   example, a function that appears in only a single call site, and does
964    ///   not otherwise escape, is often beneficial to inline regardless of its
965    ///   size (assuming we can then GC away the non-inlined version of the
966    ///   function, which we do not currently attempt to do).
967    ///
968    /// * Another potentially promising hint would be whether any of the call
969    ///   site's actual arguments are constants.
970    ///
971    /// * A general improvement would be removing the decision-tree style of
972    ///   control flow below and replacing it with (1) a pure estimated-benefit
973    ///   formula and (2) a benefit threshold. Whenever the estimated benefit
974    ///   reaches the threshold, we would inline the call. Both the formula and
975    ///   the threshold would be parameterized by tunables. This would
976    ///   effectively allow reprioritizing the relative importance of different
977    ///   hint sources, rather than being stuck with the sequence hard-coded in
978    ///   the decision tree below.
979    fn should_inline(
980        InlineHeuristicParams {
981            tunables,
982            caller_size,
983            caller_module,
984            caller_def_func,
985            caller_needs_gc_heap,
986            callee_size,
987            callee_module,
988            callee_def_func,
989            callee_needs_gc_heap,
990        }: InlineHeuristicParams,
991    ) -> bool {
992        log::trace!(
993            "considering inlining:\n\
994             \tcaller = ({caller_module:?}, {caller_def_func:?})\n\
995             \t\tsize = {caller_size}\n\
996             \t\tneeds_gc_heap = {caller_needs_gc_heap}\n\
997             \tcallee = ({callee_module:?}, {callee_def_func:?})\n\
998             \t\tsize = {callee_size}\n\
999             \t\tneeds_gc_heap = {callee_needs_gc_heap}"
1000        );
1001
1002        debug_assert!(
1003            tunables.inlining,
1004            "shouldn't even call this method if we aren't configured for inlining"
1005        );
1006        debug_assert!(
1007            caller_module != callee_module || caller_def_func != callee_def_func,
1008            "we never inline recursion"
1009        );
1010
1011        // Consider whether this is an intra-module call.
1012        //
1013        // Inlining within a single core module has most often already been done
1014        // by the toolchain that produced the module, e.g. LLVM, and any extant
1015        // function calls to small callees were presumably annotated with the
1016        // equivalent of `#[inline(never)]` or `#[cold]` but we don't have that
1017        // information anymore.
1018        if caller_module == callee_module {
1019            match tunables.inlining_intra_module {
1020                IntraModuleInlining::Yes => {}
1021
1022                IntraModuleInlining::WhenUsingGc
1023                    if caller_needs_gc_heap || callee_needs_gc_heap => {}
1024
1025                IntraModuleInlining::WhenUsingGc => {
1026                    log::trace!("  --> not inlining: intra-module call that does not use GC");
1027                    return false;
1028                }
1029
1030                IntraModuleInlining::No => {
1031                    log::trace!("  --> not inlining: intra-module call");
1032                    return false;
1033                }
1034            }
1035        }
1036
1037        // Small callees are often worth inlining regardless of the size of the
1038        // caller.
1039        if callee_size <= tunables.inlining_small_callee_size {
1040            log::trace!(
1041                "  --> inlining: callee's size is less than the small-callee size: \
1042                 {callee_size} <= {}",
1043                tunables.inlining_small_callee_size
1044            );
1045            return true;
1046        }
1047
1048        // It is often not worth inlining if the sum of the caller and callee
1049        // sizes is too large.
1050        let sum_size = caller_size.saturating_add(callee_size);
1051        if sum_size > tunables.inlining_sum_size_threshold {
1052            log::trace!(
1053                "  --> not inlining: the sum of the caller's and callee's sizes is greater than \
1054                 the inlining-sum-size threshold: {callee_size} + {caller_size} > {}",
1055                tunables.inlining_sum_size_threshold
1056            );
1057            return false;
1058        }
1059
1060        log::trace!("  --> inlining: did not find a reason we should not");
1061        true
1062    }
1063}

1065fn compile_required_builtins(engine: &Engine, raw_outputs: &mut Vec<CompileOutput>) -> Result<()> {
1066    let compiler = engine.compiler();
1067    let mut builtins = HashSet::new();
1068    let mut new_inputs: Vec<CompileInput<'_>> = Vec::new();
1069
1070    let compile_builtin = |builtin: BuiltinFunctionIndex| {
1071        Box::new(move |compiler: &dyn Compiler| {
1072            let symbol = format!("wasmtime_builtin_{}", builtin.name());
1073            let mut trampoline = compiler
1074                .compile_wasm_to_builtin(builtin, &symbol)
1075                .with_context(|| format!("failed to compile `{symbol}`"))?;
1076            if let Some(compiler) = compiler.inlining_compiler() {
1077                compiler.finish_compiling(&mut trampoline, None, &symbol)?;
1078            }
1079            Ok(CompileOutput {
1080                key: CompileKey::wasm_to_builtin_trampoline(builtin),
1081                function: CompiledFunction::Function(trampoline),
1082                symbol,
1083                start_srcloc: FilePos::default(),
1084                translation: None,
1085                func_body: None,
1086            })
1087        })
1088    };
1089
1090    for output in raw_outputs.iter() {
1091        let f = match &output.function {
1092            CompiledFunction::Function(f) => f,
1093            #[cfg(feature = "component-model")]
1094            CompiledFunction::AllCallFunc(_) => continue,
1095        };
1096        for reloc in compiler.compiled_function_relocation_targets(&*f.code) {
1097            match reloc {
1098                RelocationTarget::Builtin(i) => {
1099                    if builtins.insert(i) {
1100                        new_inputs.push(compile_builtin(i));
1101                    }
1102                }
1103                _ => {}
1104            }
1105        }
1106    }
1107    raw_outputs.extend(engine.run_maybe_parallel(new_inputs, |c| c(compiler))?);
1108    Ok(())
1109}

1111#[derive(Default)]
1112struct UnlinkedCompileOutputs<'a> {
1113    // A map from kind to `CompileOutput`.
1114    outputs: BTreeMap<CompileKind, Vec<CompileOutput<'a>>>,
1115}
1116
1117impl UnlinkedCompileOutputs<'_> {
1118    /// Flatten all our functions into a single list and remember each of their
1119    /// indices within it.
1120    fn pre_link(self) -> PreLinkOutput {
1121        // The order the functions end up within `compiled_funcs` is the order
1122        // that they will be laid out in the ELF file, so try and group hot and
1123        // cold functions together as best we can. However, because we bucket by
1124        // kind, we shouldn't have any issues with, e.g., cold trampolines
1125        // appearing in between hot Wasm functions.
1126        let mut compiled_funcs = vec![];
1127        let mut indices = FunctionIndices::default();
1128        let mut needs_gc_heap = false;
1129
1130        for output in self.outputs.into_iter().flat_map(|(_kind, outs)| outs) {
1131            let index = match output.function {
1132                CompiledFunction::Function(f) => {
1133                    needs_gc_heap |= f.needs_gc_heap;
1134                    let index = compiled_funcs.len();
1135                    compiled_funcs.push((output.symbol, f.code));
1136                    CompiledFunction::Function(index)
1137                }
1138                #[cfg(feature = "component-model")]
1139                CompiledFunction::AllCallFunc(wasmtime_environ::component::AllCallFunc {
1140                    wasm_call,
1141                    array_call,
1142                }) => {
1143                    needs_gc_heap |= array_call.needs_gc_heap;
1144                    let array_call_idx = compiled_funcs.len();
1145                    compiled_funcs.push((format!("{}_array_call", output.symbol), array_call.code));
1146
1147                    needs_gc_heap |= wasm_call.needs_gc_heap;
1148                    let wasm_call_idx = compiled_funcs.len();
1149                    compiled_funcs.push((format!("{}_wasm_call", output.symbol), wasm_call.code));
1150
1151                    CompiledFunction::AllCallFunc(wasmtime_environ::component::AllCallFunc {
1152                        array_call: array_call_idx,
1153                        wasm_call: wasm_call_idx,
1154                    })
1155                }
1156            };
1157
1158            if output.key.kind() == CompileKind::WasmFunction
1159                || output.key.kind() == CompileKind::ArrayToWasmTrampoline
1160            {
1161                indices
1162                    .compiled_func_index_to_module
1163                    .insert(index.unwrap_function(), output.key.module());
1164                indices
1165                    .start_srclocs
1166                    .insert(output.key, output.start_srcloc);
1167            }
1168
1169            indices
1170                .indices
1171                .entry(output.key.kind())
1172                .or_default()
1173                .insert(output.key, index);
1174        }
1175
1176        PreLinkOutput {
1177            needs_gc_heap,
1178            compiled_funcs,
1179            indices,
1180        }
1181    }
1182}

1184/// Our pre-link functions that have been flattened into a single list.
1185struct PreLinkOutput {
1186    /// Whether or not any of these functions require a GC heap
1187    needs_gc_heap: bool,
1188    /// The flattened list of (symbol name, compiled function) pairs, as they
1189    /// will be laid out in the object file.
1190    compiled_funcs: Vec<(String, Box<dyn Any + Send + Sync>)>,
1191    /// The `FunctionIndices` mapping our function keys to indices in that flat
1192    /// list.
1193    indices: FunctionIndices,
1194}
1195
1196#[derive(Default)]
1197struct FunctionIndices {
1198    // A reverse map from an index in `compiled_funcs` to the
1199    // `StaticModuleIndex` for that function.
1200    compiled_func_index_to_module: HashMap<usize, StaticModuleIndex>,
1201
1202    // A map of wasm functions and where they're located in the original file.
1203    start_srclocs: HashMap<CompileKey, FilePos>,
1204
1205    // The index of each compiled function, bucketed by compile key kind.
1206    indices: BTreeMap<CompileKind, BTreeMap<CompileKey, CompiledFunction<usize>>>,
1207}
1208
1209impl FunctionIndices {
1210    /// Link the compiled functions together, resolving relocations, and append
1211    /// them to the given ELF file.
1212    fn link_and_append_code<'a>(
1213        mut self,
1214        types: &ModuleTypesBuilder,
1215        mut obj: object::write::Object<'static>,
1216        engine: &'a Engine,
1217        compiled_funcs: Vec<(String, Box<dyn Any + Send + Sync>)>,
1218        translations: PrimaryMap<StaticModuleIndex, ModuleTranslation<'_>>,
1219        dwarf_package_bytes: Option<&[u8]>,
1220    ) -> Result<(wasmtime_environ::ObjectBuilder<'a>, Artifacts)> {
1221        // Append all the functions to the ELF file.
1222        //
1223        // The result is a vector parallel to `compiled_funcs` where
1224        // `symbol_ids_and_locs[i]` is the symbol ID and function location of
1225        // `compiled_funcs[i]`.
1226        let compiler = engine.compiler();
1227        let tunables = engine.tunables();
1228        let symbol_ids_and_locs = compiler.append_code(
1229            &mut obj,
1230            &compiled_funcs,
1231            &|caller_index: usize, callee: RelocationTarget| match callee {
1232                RelocationTarget::Wasm(callee_index) => {
1233                    let caller_module = self
1234                        .compiled_func_index_to_module
1235                        .get(&caller_index)
1236                        .copied()
1237                        .expect("should only reloc inside wasm function callers");
1238                    let key = if let Some(def_func_index) = translations[caller_module]
1239                        .module
1240                        .defined_func_index(callee_index)
1241                    {
1242                        CompileKey::wasm_function(caller_module, def_func_index)
1243                    } else {
1244                        let (def_module, def_func_index) = translations[caller_module]
1245                            .known_imported_functions[callee_index]
1246                            .expect(
1247                                "a direct call to an imported function must have a \
1248                                 statically-known import",
1249                            );
1250                        CompileKey::wasm_function(def_module, def_func_index)
1251                    };
1252                    self.indices[&CompileKind::WasmFunction][&key].unwrap_function()
1253                }
1254                RelocationTarget::Builtin(builtin) => self.indices
1255                    [&CompileKind::WasmToBuiltinTrampoline]
1256                    [&CompileKey::wasm_to_builtin_trampoline(builtin)]
1257                    .unwrap_function(),
1258                RelocationTarget::PulleyHostcall(_) => {
1259                    unreachable!("relocation is resolved at runtime, not compile time");
1260                }
1261            },
1262        )?;
1263
1264        // If requested, generate and add DWARF information.
1265        if tunables.generate_native_debuginfo {
1266            compiler.append_dwarf(
1267                &mut obj,
1268                &translations,
1269                &|module, func| {
1270                    let bucket = &self.indices[&CompileKind::WasmFunction];
1271                    let i = bucket[&CompileKey::wasm_function(module, func)].unwrap_function();
1272                    (symbol_ids_and_locs[i].0, &*compiled_funcs[i].1)
1273                },
1274                dwarf_package_bytes,
1275                tunables,
1276            )?;
1277        }
1278
1279        let mut obj = wasmtime_environ::ObjectBuilder::new(obj, tunables);
1280        let mut artifacts = Artifacts::default();
1281
1282        // Remove this as it's not needed by anything below and we'll debug
1283        // assert `self.indices` is empty, so this is acknowledgement that this
1284        // is a pure runtime implementation detail and not needed in any
1285        // metadata generated below.
1286        self.indices.remove(&CompileKind::WasmToBuiltinTrampoline);
1287
1288        // Finally, build our binary artifacts that map things like `FuncIndex`
1289        // to a function location and all of that using the indices we saved
1290        // earlier and the function locations we just received after appending
1291        // the code.
1292
1293        let mut wasm_functions = self
1294            .indices
1295            .remove(&CompileKind::WasmFunction)
1296            .unwrap_or_default()
1297            .into_iter()
1298            .peekable();
1299
1300        fn wasm_functions_for_module(
1301            wasm_functions: &mut std::iter::Peekable<
1302                btree_map::IntoIter<CompileKey, CompiledFunction<usize>>,
1303            >,
1304            module: StaticModuleIndex,
1305        ) -> impl Iterator<Item = (CompileKey, CompiledFunction<usize>)> + '_ {
1306            std::iter::from_fn(move || {
1307                let (key, _) = wasm_functions.peek()?;
1308                if key.module() == module {
1309                    wasm_functions.next()
1310                } else {
1311                    None
1312                }
1313            })
1314        }
1315
1316        let mut array_to_wasm_trampolines = self
1317            .indices
1318            .remove(&CompileKind::ArrayToWasmTrampoline)
1319            .unwrap_or_default();
1320
1321        // NB: unlike the above maps this is not emptied out during iteration
1322        // since each module may reach into different portions of this map.
1323        let wasm_to_array_trampolines = self
1324            .indices
1325            .remove(&CompileKind::WasmToArrayTrampoline)
1326            .unwrap_or_default();
1327
1328        artifacts.modules = translations
1329            .into_iter()
1330            .map(|(module, mut translation)| {
1331                // If configured attempt to use static memory initialization which
1332                // can either at runtime be implemented as a single memcpy to
1333                // initialize memory or otherwise enabling virtual-memory-tricks
1334                // such as mmap'ing from a file to get copy-on-write.
1335                if engine.tunables().memory_init_cow {
1336                    let align = compiler.page_size_align();
1337                    let max_always_allowed = engine.config().memory_guaranteed_dense_image_size;
1338                    translation.try_static_init(align, max_always_allowed);
1339                }
1340
1341                // Attempt to convert table initializer segments to
1342                // FuncTable representation where possible, to enable
1343                // table lazy init.
1344                if engine.tunables().table_lazy_init {
1345                    translation.try_func_table_init();
1346                }
1347
1348                let funcs: PrimaryMap<DefinedFuncIndex, CompiledFunctionInfo> =
1349                    wasm_functions_for_module(&mut wasm_functions, module)
1350                        .map(|(key, wasm_func_index)| {
1351                            let wasm_func_index = wasm_func_index.unwrap_function();
1352                            let wasm_func_loc = symbol_ids_and_locs[wasm_func_index].1;
1353                            let start_srcloc = self.start_srclocs.remove(&key).unwrap();
1354
1355                            let array_to_wasm_trampoline = array_to_wasm_trampolines
1356                                .remove(&CompileKey::array_to_wasm_trampoline(
1357                                    key.module(),
1358                                    DefinedFuncIndex::from_u32(key.index),
1359                                ))
1360                                .map(|x| symbol_ids_and_locs[x.unwrap_function()].1);
1361
1362                            CompiledFunctionInfo {
1363                                start_srcloc,
1364                                wasm_func_loc,
1365                                array_to_wasm_trampoline,
1366                            }
1367                        })
1368                        .collect();
1369
1370                let unique_and_sorted_trampoline_sigs = translation
1371                    .module
1372                    .types
1373                    .iter()
1374                    .map(|(_, ty)| ty.unwrap_module_type_index())
1375                    .filter(|idx| types[*idx].is_func())
1376                    .map(|idx| types.trampoline_type(idx))
1377                    .collect::<BTreeSet<_>>();
1378                let wasm_to_array_trampolines = unique_and_sorted_trampoline_sigs
1379                    .iter()
1380                    .map(|idx| {
1381                        let trampoline = types.trampoline_type(*idx);
1382                        let key = CompileKey::wasm_to_array_trampoline(trampoline);
1383                        let compiled = wasm_to_array_trampolines[&key];
1384                        (*idx, symbol_ids_and_locs[compiled.unwrap_function()].1)
1385                    })
1386                    .collect();
1387
1388                obj.append(translation, funcs, wasm_to_array_trampolines)
1389            })
1390            .collect::<Result<PrimaryMap<_, _>>>()?;
1391
1392        #[cfg(feature = "component-model")]
1393        {
1394            artifacts.trampolines = self
1395                .indices
1396                .remove(&CompileKind::Trampoline)
1397                .unwrap_or_default()
1398                .into_iter()
1399                .map(|(_id, x)| x.unwrap_all_call_func().map(|i| symbol_ids_and_locs[i].1))
1400                .collect();
1401            let map = self
1402                .indices
1403                .remove(&CompileKind::ResourceDropWasmToArrayTrampoline)
1404                .unwrap_or_default();
1405            assert!(map.len() <= 1);
1406            artifacts.resource_drop_wasm_to_array_trampoline = map
1407                .into_iter()
1408                .next()
1409                .map(|(_id, x)| symbol_ids_and_locs[x.unwrap_function()].1);
1410        }
1411
1412        debug_assert!(
1413            self.indices.is_empty(),
1414            "Should have processed all compile outputs"
1415        );
1416
1417        Ok((obj, artifacts))
1418    }
1419}
1420
1421/// The artifacts necessary for finding and calling Wasm functions at runtime,
1422/// to be serialized into an ELF file.
1423#[derive(Default)]
1424struct Artifacts {
1425    modules: PrimaryMap<StaticModuleIndex, CompiledModuleInfo>,
1426    #[cfg(feature = "component-model")]
1427    trampolines: PrimaryMap<
1428        wasmtime_environ::component::TrampolineIndex,
1429        wasmtime_environ::component::AllCallFunc<wasmtime_environ::FunctionLoc>,
1430    >,
1431    #[cfg(feature = "component-model")]
1432    resource_drop_wasm_to_array_trampoline: Option<wasmtime_environ::FunctionLoc>,
1433}
1434
1435impl Artifacts {
1436    /// Assuming this compilation was for a single core Wasm module, get the
1437    /// resulting `CompiledModuleInfo`.
1438    fn unwrap_as_module_info(self) -> CompiledModuleInfo {
1439        assert_eq!(self.modules.len(), 1);
1440        #[cfg(feature = "component-model")]
1441        assert!(self.trampolines.is_empty());
1442        self.modules.into_iter().next().unwrap().1
1443    }
1444}

1446/// Extend `dest` with `items` and return the range of indices in `dest` where
1447/// they ended up.
1448fn extend_with_range<T>(dest: &mut Vec<T>, items: impl IntoIterator<Item = T>) -> Range<u32> {
1449    let start = dest.len();
1450    let start = u32::try_from(start).unwrap();
1451
1452    dest.extend(items);
1453
1454    let end = dest.len();
1455    let end = u32::try_from(end).unwrap();
1456
1457    start..end
1458}
