95impl Drop for Compiler {
96    fn drop(&mut self) {
97        if self.cache_store.is_none() {
98            return;
99        }
100
101        let mut num_hits = 0;
102        let mut num_cached = 0;
103        for ctx in self.contexts.lock().unwrap().iter() {
104            if let Some(ref cache_ctx) = ctx.incremental_cache_ctx {
105                num_hits += cache_ctx.num_hits;
106                num_cached += cache_ctx.num_cached;
107            }
108        }
109
110        let total = num_hits + num_cached;
111        if num_hits + num_cached > 0 {
112            log::trace!(
113                "Incremental compilation cache stats: {}/{} = {}% (hits/lookup)\ncached: {}",
114                num_hits,
115                total,
116                (num_hits as f32) / (total as f32) * 100.0,
117                num_cached
118            );
119        }
120    }
121}

145    /// Perform an indirect call from Cranelift-generated code to native code in
146    /// Wasmtime itself.
147    ///
148    /// For native platforms this is a simple `call_indirect` instruction but
149    /// for the Pulley backend this is special as it's transitioning from
150    /// Cranelift-generated bytecode to native code on the host. That requires a
151    /// special opcode in the interpreter and is modeled slightly differently in
152    /// Cranelift IR.
153    fn call_indirect_host(
154        &self,
155        builder: &mut FunctionBuilder<'_>,
156        hostcall: impl Into<HostCall>,
157        sig: ir::SigRef,
158        addr: Value,
159        args: &[Value],
160    ) -> ir::Inst {
161        let signature = &builder.func.dfg.signatures[sig];
162
163        // When calling the host we should always be using the platform's
164        // default calling convention since it'll be calling Rust code in
165        // Wasmtime itself.
166        assert_eq!(signature.call_conv, self.isa.default_call_conv());
167
168        // If this target is actually pulley then the goal is to emit the custom
169        // `call_indirect_host` pulley opcode. That's encoded in Cranelift as a
170        // `call` instruction where the name is `colocated: false`. This will
171        // force a pulley-specific relocation to get emitted in addition to
172        // using the `call_indirect_host` instruction.
173        if self.isa.triple().is_pulley() {
174            let mut new_signature = signature.clone();
175            new_signature
176                .params
177                .insert(0, ir::AbiParam::new(self.isa.pointer_type()));
178            let new_sig = builder.func.import_signature(new_signature);
179            let name = ir::ExternalName::User(builder.func.declare_imported_user_function(
180                ir::UserExternalName {
181                    namespace: crate::NS_PULLEY_HOSTCALL,
182                    index: hostcall.into().index(),
183                },
184            ));
185            let func = builder.func.import_function(ir::ExtFuncData {
186                name,
187                signature: new_sig,
188                // This is the signal that a special `call_indirect_host`
189                // opcode is used to jump from pulley to the host.
190                colocated: false,
191            });
192            let mut raw_args = vec![addr];
193            raw_args.extend_from_slice(args);
194            return builder.ins().call(func, &raw_args);
195        }
196
197        builder.ins().call_indirect(sig, addr, args)
198    }

206    fn compile_function(
207        &self,
208        translation: &ModuleTranslation<'_>,
209        func_index: DefinedFuncIndex,
210        input: FunctionBodyData<'_>,
211        types: &ModuleTypesBuilder,
212        symbol: &str,
213    ) -> Result<CompiledFunctionBody, CompileError> {
214        let isa = &*self.isa;
215        let module = &translation.module;
216        let func_index = module.func_index(func_index);
217        let sig = translation.module.functions[func_index]
218            .signature
219            .unwrap_module_type_index();
220        let wasm_func_ty = types[sig].unwrap_func();
221
222        let mut compiler = self.function_compiler();
223
224        let context = &mut compiler.cx.codegen_context;
225        context.func.signature = wasm_call_signature(isa, wasm_func_ty, &self.tunables);
226        context.func.name = UserFuncName::User(UserExternalName {
227            namespace: crate::NS_WASM_FUNC,
228            index: func_index.as_u32(),
229        });
230
231        if self.tunables.generate_native_debuginfo {
232            context.func.collect_debug_info();
233        }
234
235        let mut func_env = FuncEnvironment::new(self, translation, types, wasm_func_ty);
236
237        // The `stack_limit` global value below is the implementation of stack
238        // overflow checks in Wasmtime.
239        //
240        // The Wasm spec defines that stack overflows will raise a trap, and
241        // there's also an added constraint where as an embedder you frequently
242        // are running host-provided code called from wasm. WebAssembly and
243        // native code currently share the same call stack, so Wasmtime needs to
244        // make sure that host-provided code will have enough call-stack
245        // available to it.
246        //
247        // The way that stack overflow is handled here is by adding a prologue
248        // check to all functions for how much native stack is remaining. The
249        // `VMContext` pointer is the first argument to all functions, and the
250        // first field of this structure is `*const VMStoreContext` and the
251        // third field of that is the stack limit. Note that the stack limit in
252        // this case means "if the stack pointer goes below this, trap". Each
253        // function which consumes stack space or isn't a leaf function starts
254        // off by loading the stack limit, checking it against the stack
255        // pointer, and optionally traps.
256        //
257        // This manual check allows the embedder to give wasm a relatively
258        // precise amount of stack allocation. Using this scheme we reserve a
259        // chunk of stack for wasm code relative from where wasm code was
260        // called. This ensures that native code called by wasm should have
261        // native stack space to run, and the numbers of stack spaces here
262        // should all be configurable for various embeddings.
263        //
264        // Note that this check is independent of each thread's stack guard page
265        // here. If the stack guard page is reached that's still considered an
266        // abort for the whole program since the runtime limits configured by
267        // the embedder should cause wasm to trap before it reaches that
268        // (ensuring the host has enough space as well for its functionality).
269        if !isa.triple().is_pulley() {
270            let vmctx = context
271                .func
272                .create_global_value(ir::GlobalValueData::VMContext);
273            let interrupts_ptr = context.func.create_global_value(ir::GlobalValueData::Load {
274                base: vmctx,
275                offset: i32::from(func_env.offsets.ptr.vmctx_store_context()).into(),
276                global_type: isa.pointer_type(),
277                flags: MemFlags::trusted().with_readonly(),
278            });
279            let stack_limit = context.func.create_global_value(ir::GlobalValueData::Load {
280                base: interrupts_ptr,
281                offset: i32::from(func_env.offsets.ptr.vmstore_context_stack_limit()).into(),
282                global_type: isa.pointer_type(),
283                flags: MemFlags::trusted(),
284            });
285            if self.tunables.signals_based_traps {
286                context.func.stack_limit = Some(stack_limit);
287            } else {
288                func_env.stack_limit_at_function_entry = Some(stack_limit);
289            }
290        }
291        let FunctionBodyData { validator, body } = input;
292        let mut validator =
293            validator.into_validator(mem::take(&mut compiler.cx.validator_allocations));
294        compiler.cx.func_translator.translate_body(
295            &mut validator,
296            body.clone(),
297            &mut context.func,
298            &mut func_env,
299        )?;
300
301        if self.tunables.inlining {
302            compiler
303                .cx
304                .codegen_context
305                .legalize(isa)
306                .map_err(|e| CompileError::Codegen(e.to_string()))?;
307        }
308
309        let timing = cranelift_codegen::timing::take_current();
310        log::debug!("`{symbol}` translated to CLIF in {:?}", timing.total());
311        log::trace!("`{symbol}` timing info\n{timing}");
312
313        Ok(CompiledFunctionBody {
314            code: Box::new(Some(compiler.cx)),
315            needs_gc_heap: func_env.needs_gc_heap(),
316        })
317    }

319    fn compile_array_to_wasm_trampoline(
320        &self,
321        translation: &ModuleTranslation<'_>,
322        types: &ModuleTypesBuilder,
323        def_func_index: DefinedFuncIndex,
324        _symbol: &str,
325    ) -> Result<CompiledFunctionBody, CompileError> {
326        let func_index = translation.module.func_index(def_func_index);
327        let sig = translation.module.functions[func_index]
328            .signature
329            .unwrap_module_type_index();
330        let wasm_func_ty = types[sig].unwrap_func();
331
332        let isa = &*self.isa;
333        let pointer_type = isa.pointer_type();
334        let wasm_call_sig = wasm_call_signature(isa, wasm_func_ty, &self.tunables);
335        let array_call_sig = array_call_signature(isa);
336
337        let mut compiler = self.function_compiler();
338        let func = ir::Function::with_name_signature(Default::default(), array_call_sig);
339        let (mut builder, block0) = compiler.builder(func);
340
341        let (vmctx, caller_vmctx, values_vec_ptr, values_vec_len) = {
342            let params = builder.func.dfg.block_params(block0);
343            (params[0], params[1], params[2], params[3])
344        };
345
346        // First load the actual arguments out of the array.
347        let mut args = self.load_values_from_array(
348            wasm_func_ty.params(),
349            &mut builder,
350            values_vec_ptr,
351            values_vec_len,
352        );
353        args.insert(0, caller_vmctx);
354        args.insert(0, vmctx);
355
356        // Just before we enter Wasm, save our stack pointer.
357        //
358        // Assert that we were really given a core Wasm vmctx, since that's
359        // what we are assuming with our offsets below.
360        debug_assert_vmctx_kind(isa, &mut builder, vmctx, wasmtime_environ::VMCONTEXT_MAGIC);
361        let offsets = VMOffsets::new(isa.pointer_bytes(), &translation.module);
362        let vm_store_context_offset = offsets.ptr.vmctx_store_context();
363        save_last_wasm_entry_fp(
364            &mut builder,
365            pointer_type,
366            &offsets.ptr,
367            vm_store_context_offset.into(),
368            vmctx,
369        );
370
371        // Then call the Wasm function with those arguments.
372        let call = declare_and_call(&mut builder, wasm_call_sig, func_index.as_u32(), &args);
373        let results = builder.func.dfg.inst_results(call).to_vec();
374
375        // Then store the results back into the array.
376        self.store_values_to_array(
377            &mut builder,
378            wasm_func_ty.returns(),
379            &results,
380            values_vec_ptr,
381            values_vec_len,
382        );
383
384        // At this time wasm functions always signal traps with longjmp or some
385        // similar sort of routine, so if we got this far that means that the
386        // function did not trap, so return a "true" value here to indicate that
387        // to satisfy the ABI of the array-call signature.
388        let true_return = builder.ins().iconst(ir::types::I8, 1);
389        builder.ins().return_(&[true_return]);
390        builder.finalize();
391
392        Ok(CompiledFunctionBody {
393            code: Box::new(Some(compiler.cx)),
394            needs_gc_heap: false,
395        })
396    }

398    fn compile_wasm_to_array_trampoline(
399        &self,
400        wasm_func_ty: &WasmFuncType,
401        _symbol: &str,
402    ) -> Result<CompiledFunctionBody, CompileError> {
403        let isa = &*self.isa;
404        let pointer_type = isa.pointer_type();
405        let wasm_call_sig = wasm_call_signature(isa, wasm_func_ty, &self.tunables);
406        let array_call_sig = array_call_signature(isa);
407
408        let mut compiler = self.function_compiler();
409        let func = ir::Function::with_name_signature(Default::default(), wasm_call_sig);
410        let (mut builder, block0) = compiler.builder(func);
411
412        let args = builder.func.dfg.block_params(block0).to_vec();
413        let callee_vmctx = args[0];
414        let caller_vmctx = args[1];
415
416        // We are exiting Wasm, so save our PC and FP.
417        //
418        // Assert that the caller vmctx really is a core Wasm vmctx, since
419        // that's what we are assuming with our offsets below.
420        debug_assert_vmctx_kind(
421            isa,
422            &mut builder,
423            caller_vmctx,
424            wasmtime_environ::VMCONTEXT_MAGIC,
425        );
426        let ptr = isa.pointer_bytes();
427        let vm_store_context = builder.ins().load(
428            pointer_type,
429            MemFlags::trusted(),
430            caller_vmctx,
431            i32::from(ptr.vmcontext_store_context()),
432        );
433        save_last_wasm_exit_fp_and_pc(&mut builder, pointer_type, &ptr, vm_store_context);
434
435        // Spill all wasm arguments to the stack in `ValRaw` slots.
436        let (args_base, args_len) =
437            self.allocate_stack_array_and_spill_args(wasm_func_ty, &mut builder, &args[2..]);
438        let args_len = builder.ins().iconst(pointer_type, i64::from(args_len));
439
440        // Load the actual callee out of the
441        // `VMArrayCallHostFuncContext::host_func`.
442        let ptr_size = isa.pointer_bytes();
443        let callee = builder.ins().load(
444            pointer_type,
445            MemFlags::trusted(),
446            callee_vmctx,
447            ptr_size.vmarray_call_host_func_context_func_ref() + ptr_size.vm_func_ref_array_call(),
448        );
449
450        // Do an indirect call to the callee.
451        let callee_signature = builder.func.import_signature(array_call_sig);
452        let call = self.call_indirect_host(
453            &mut builder,
454            HostCall::ArrayCall,
455            callee_signature,
456            callee,
457            &[callee_vmctx, caller_vmctx, args_base, args_len],
458        );
459        let succeeded = builder.func.dfg.inst_results(call)[0];
460        self.raise_if_host_trapped(&mut builder, caller_vmctx, succeeded);
461        let results =
462            self.load_values_from_array(wasm_func_ty.returns(), &mut builder, args_base, args_len);
463        builder.ins().return_(&results);
464        builder.finalize();
465
466        Ok(CompiledFunctionBody {
467            code: Box::new(Some(compiler.cx)),
468            needs_gc_heap: false,
469        })
470    }

472    fn append_code(
473        &self,
474        obj: &mut Object<'static>,
475        funcs: &[(String, Box<dyn Any + Send + Sync>)],
476        resolve_reloc: &dyn Fn(usize, RelocationTarget) -> usize,
477    ) -> Result<Vec<(SymbolId, FunctionLoc)>> {
478        let mut builder =
479            ModuleTextBuilder::new(obj, self, self.isa.text_section_builder(funcs.len()));
480        if self.linkopts.force_jump_veneers {
481            builder.force_veneers();
482        }
483        let mut addrs = AddressMapSection::default();
484        let mut traps = TrapEncodingBuilder::default();
485        let mut stack_maps = StackMapSection::default();
486
487        let mut ret = Vec::with_capacity(funcs.len());
488        for (i, (sym, func)) in funcs.iter().enumerate() {
489            let func = func.downcast_ref::<CompiledFunction>().unwrap();
490            let (sym, range) = builder.append_func(&sym, func, |idx| resolve_reloc(i, idx));
491            if self.tunables.generate_address_map {
492                let addr = func.address_map();
493                addrs.push(range.clone(), &addr.instructions);
494            }
495            clif_to_env_stack_maps(
496                &mut stack_maps,
497                range.clone(),
498                func.buffer.user_stack_maps(),
499            );
500            traps.push(range.clone(), &func.traps().collect::<Vec<_>>());
501            builder.append_padding(self.linkopts.padding_between_functions);
502            let info = FunctionLoc {
503                start: u32::try_from(range.start).unwrap(),
504                length: u32::try_from(range.end - range.start).unwrap(),
505            };
506            ret.push((sym, info));
507        }
508
509        builder.finish();
510
511        if self.tunables.generate_address_map {
512            addrs.append_to(obj);
513        }
514        stack_maps.append_to(obj);
515        traps.append_to(obj);
516
517        Ok(ret)
518    }

541    fn append_dwarf<'a>(
542        &self,
543        obj: &mut Object<'_>,
544        translations: &'a PrimaryMap<StaticModuleIndex, ModuleTranslation<'a>>,
545        get_func: &'a dyn Fn(
546            StaticModuleIndex,
547            DefinedFuncIndex,
548        ) -> (SymbolId, &'a (dyn Any + Send + Sync)),
549        dwarf_package_bytes: Option<&'a [u8]>,
550        tunables: &'a Tunables,
551    ) -> Result<()> {
552        let get_func = move |m, f| {
553            let (sym, any) = get_func(m, f);
554            (
555                sym,
556                any.downcast_ref::<CompiledFunction>().unwrap().metadata(),
557            )
558        };
559        let mut compilation = crate::debug::Compilation::new(
560            &*self.isa,
561            translations,
562            &get_func,
563            dwarf_package_bytes,
564            tunables,
565        );
566        let dwarf_sections = crate::debug::emit_dwarf(&*self.isa, &mut compilation)
567            .with_context(|| "failed to emit DWARF debug information")?;
568
569        let (debug_bodies, debug_relocs): (Vec<_>, Vec<_>) = dwarf_sections
570            .iter()
571            .map(|s| ((s.name, &s.body), (s.name, &s.relocs)))
572            .unzip();
573        let mut dwarf_sections_ids = HashMap::new();
574        for (name, body) in debug_bodies {
575            let segment = obj.segment_name(StandardSegment::Debug).to_vec();
576            let section_id = obj.add_section(segment, name.as_bytes().to_vec(), SectionKind::Debug);
577            dwarf_sections_ids.insert(name, section_id);
578            obj.append_section_data(section_id, &body, 1);
579        }
580
581        // Write all debug data relocations.
582        for (name, relocs) in debug_relocs {
583            let section_id = *dwarf_sections_ids.get(name).unwrap();
584            for reloc in relocs {
585                let target_symbol = match reloc.target {
586                    DwarfSectionRelocTarget::Func(id) => compilation.symbol_id(id),
587                    DwarfSectionRelocTarget::Section(name) => {
588                        obj.section_symbol(dwarf_sections_ids[name])
589                    }
590                };
591                obj.add_relocation(
592                    section_id,
593                    object::write::Relocation {
594                        offset: u64::from(reloc.offset),
595                        symbol: target_symbol,
596                        addend: i64::from(reloc.addend),
597                        flags: RelocationFlags::Generic {
598                            size: reloc.size << 3,
599                            kind: RelocationKind::Absolute,
600                            encoding: RelocationEncoding::Generic,
601                        },
602                    },
603                )?;
604            }
605        }
606
607        Ok(())
608    }

614    fn compile_wasm_to_builtin(
615        &self,
616        index: BuiltinFunctionIndex,
617        _symbol: &str,
618    ) -> Result<CompiledFunctionBody, CompileError> {
619        let isa = &*self.isa;
620        let ptr_size = isa.pointer_bytes();
621        let pointer_type = isa.pointer_type();
622        let sigs = BuiltinFunctionSignatures::new(self);
623        let wasm_sig = sigs.wasm_signature(index);
624        let host_sig = sigs.host_signature(index);
625
626        let mut compiler = self.function_compiler();
627        let func = ir::Function::with_name_signature(Default::default(), wasm_sig.clone());
628        let (mut builder, block0) = compiler.builder(func);
629        let vmctx = builder.block_params(block0)[0];
630
631        // Debug-assert that this is the right kind of vmctx, and then
632        // additionally perform the "routine of the exit trampoline" of saving
633        // fp/pc/etc.
634        debug_assert_vmctx_kind(isa, &mut builder, vmctx, wasmtime_environ::VMCONTEXT_MAGIC);
635        let vm_store_context = builder.ins().load(
636            pointer_type,
637            MemFlags::trusted(),
638            vmctx,
639            ptr_size.vmcontext_store_context(),
640        );
641        save_last_wasm_exit_fp_and_pc(&mut builder, pointer_type, &ptr_size, vm_store_context);
642
643        // Now it's time to delegate to the actual builtin. Forward all our own
644        // arguments to the libcall itself.
645        let args = builder.block_params(block0).to_vec();
646        let call = self.call_builtin(&mut builder, vmctx, &args, index, host_sig);
647        let results = builder.func.dfg.inst_results(call).to_vec();
648
649        // Libcalls do not explicitly `longjmp` for example but instead return a
650        // code indicating whether they trapped or not. This means that it's the
651        // responsibility of the trampoline to check for an trapping return
652        // value and raise a trap as appropriate. With the `results` above check
653        // what `index` is and for each libcall that has a trapping return value
654        // process it here.
655        match index.trap_sentinel() {
656            Some(TrapSentinel::Falsy) => {
657                self.raise_if_host_trapped(&mut builder, vmctx, results[0]);
658            }
659            Some(TrapSentinel::NegativeTwo) => {
660                let ty = builder.func.dfg.value_type(results[0]);
661                let trapped = builder.ins().iconst(ty, -2);
662                let succeeded = builder.ins().icmp(IntCC::NotEqual, results[0], trapped);
663                self.raise_if_host_trapped(&mut builder, vmctx, succeeded);
664            }
665            Some(TrapSentinel::Negative) => {
666                let ty = builder.func.dfg.value_type(results[0]);
667                let zero = builder.ins().iconst(ty, 0);
668                let succeeded =
669                    builder
670                        .ins()
671                        .icmp(IntCC::SignedGreaterThanOrEqual, results[0], zero);
672                self.raise_if_host_trapped(&mut builder, vmctx, succeeded);
673            }
674            Some(TrapSentinel::NegativeOne) => {
675                let ty = builder.func.dfg.value_type(results[0]);
676                let minus_one = builder.ins().iconst(ty, -1);
677                let succeeded = builder.ins().icmp(IntCC::NotEqual, results[0], minus_one);
678                self.raise_if_host_trapped(&mut builder, vmctx, succeeded);
679            }
680            None => {}
681        }
682
683        // And finally, return all the results of this libcall.
684        builder.ins().return_(&results);
685        builder.finalize();
686
687        Ok(CompiledFunctionBody {
688            code: Box::new(Some(compiler.cx)),
689            needs_gc_heap: false,
690        })
691    }

738        &self,
739        func_body: &mut CompiledFunctionBody,
740        get_callee: &'a mut dyn FnMut(FuncIndex) -> Option<&'a CompiledFunctionBody>,
741    ) -> Result<()> {
742        let code = func_body
743            .code
744            .downcast_mut::<Option<CompilerContext>>()

751        struct Inliner<'a>(&'a mut dyn FnMut(FuncIndex) -> Option<&'a CompiledFunctionBody>);
752
753        impl cranelift_codegen::inline::Inline for Inliner<'_> {
754            fn inline(
755                &mut self,
756                caller: &ir::Function,
757                _call_inst: ir::Inst,
758                _call_opcode: ir::Opcode,
759                callee: ir::FuncRef,
760                _call_args: &[ir::Value],
761            ) -> InlineCommand<'_> {
762                let callee = &caller.dfg.ext_funcs[callee].name;
763                let callee = match callee {
764                    ir::ExternalName::User(callee) => *callee,
765                    ir::ExternalName::TestCase(_)
766                    | ir::ExternalName::LibCall(_)
767                    | ir::ExternalName::KnownSymbol(_) => return InlineCommand::KeepCall,
768                };
769                let callee = &caller.params.user_named_funcs()[callee];
770                let callee = if callee.namespace == crate::NS_WASM_FUNC {
771                    FuncIndex::from_u32(callee.index)
772                } else {
773                    return InlineCommand::KeepCall;
774                };
775                match (self.0)(callee) {
776                    None => InlineCommand::KeepCall,
777                    Some(func_body) => {
778                        let cx = func_body
779                            .code
780                            .downcast_ref::<Option<CompilerContext>>()
781                            .unwrap();
782                        InlineCommand::Inline(Cow::Borrowed(
783                            &cx.as_ref().unwrap().codegen_context.func,
784                        ))
785                    }
786                }
787            }
788        }
789    }

791    fn finish_compiling(
792        &self,
793        func_body: &mut CompiledFunctionBody,
794        input: Option<wasmparser::FunctionBody<'_>>,
795        symbol: &str,
796    ) -> Result<()> {
797        let cx = func_body
798            .code
799            .downcast_mut::<Option<CompilerContext>>()
800            .unwrap()
801            .take()
802            .unwrap();
803        let compiler = FunctionCompiler { compiler: self, cx };
804
805        let symbol = match compiler.cx.abi {
806            None => Cow::Borrowed(symbol),
807            Some(Abi::Wasm) => Cow::Owned(format!("{symbol}_wasm_call")),
808            Some(Abi::Array) => Cow::Owned(format!("{symbol}_array_call")),
809        };
810
811        let compiled_func = if let Some(input) = input {
812            compiler.finish_with_info(Some((&input, &self.tunables)), &symbol)?
813        } else {
814            compiler.finish(&symbol)?
815        };
816
817        let timing = cranelift_codegen::timing::take_current();
818        log::debug!("`{symbol}` compiled in {:?}", timing.total());
819        log::trace!("`{symbol}` timing info\n{timing}");
820
821        func_body.code = Box::new(compiled_func);
822        Ok(())
823    }

889    /// This function will allocate a stack slot suitable for storing both the
890    /// arguments and return values of the function, and then the arguments will
891    /// all be stored in this block.
892    ///
893    /// `block0` must be the entry block of the function and `ty` must be the
894    /// Wasm function type of the trampoline.
895    ///
896    /// The stack slot pointer is returned in addition to the size, in units of
897    /// `ValRaw`, of the stack slot.
898    fn allocate_stack_array_and_spill_args(
899        &self,
900        ty: &WasmFuncType,
901        builder: &mut FunctionBuilder,
902        args: &[ir::Value],
903    ) -> (Value, u32) {
904        let isa = &*self.isa;
905        let pointer_type = isa.pointer_type();
906
907        // Compute the size of the values vector.
908        let value_size = mem::size_of::<u128>();
909        let values_vec_len = cmp::max(ty.params().len(), ty.returns().len());
910        let values_vec_byte_size = u32::try_from(value_size * values_vec_len).unwrap();
911        let values_vec_len = u32::try_from(values_vec_len).unwrap();
912
913        let slot = builder.func.create_sized_stack_slot(ir::StackSlotData::new(
914            ir::StackSlotKind::ExplicitSlot,
915            values_vec_byte_size,
916            4,
917        ));
918        let values_vec_ptr = builder.ins().stack_addr(pointer_type, slot, 0);
919
920        {
921            let values_vec_len = builder
922                .ins()
923                .iconst(ir::types::I32, i64::from(values_vec_len));
924            self.store_values_to_array(builder, ty.params(), args, values_vec_ptr, values_vec_len);
925        }
926
927        (values_vec_ptr, values_vec_len)
928    }

930    /// Store values to an array in the array calling convention.
931    ///
932    /// Used either to store arguments to the array when calling a function
933    /// using the array calling convention, or used to store results to the
934    /// array when implementing a function that exposes the array calling
935    /// convention.
936    fn store_values_to_array(
937        &self,
938        builder: &mut FunctionBuilder,
939        types: &[WasmValType],
940        values: &[Value],
941        values_vec_ptr: Value,
942        values_vec_capacity: Value,
943    ) {
944        debug_assert_eq!(types.len(), values.len());
945        debug_assert_enough_capacity_for_length(builder, types.len(), values_vec_capacity);
946
947        // Note that loads and stores are unconditionally done in the
948        // little-endian format rather than the host's native-endianness,
949        // despite this load/store being unrelated to execution in wasm itself.
950        // For more details on this see the `ValRaw` type in
951        // `wasmtime::runtime::vm`.
952        let flags = ir::MemFlags::new()
953            .with_notrap()
954            .with_endianness(ir::Endianness::Little);
955
956        let value_size = mem::size_of::<u128>();
957        for (i, val) in values.iter().copied().enumerate() {
958            crate::unbarriered_store_type_at_offset(
959                &mut builder.cursor(),
960                flags,
961                values_vec_ptr,
962                i32::try_from(i * value_size).unwrap(),
963                val,
964            );
965        }
966    }
967
968    /// Used for loading the values of an array-call host function's value
969    /// array.
970    ///
971    /// This can be used to load arguments out of the array if the trampoline we
972    /// are building exposes the array calling convention, or it can be used to
973    /// load results out of the array if the trampoline we are building calls a
974    /// function that uses the array calling convention.
975    fn load_values_from_array(
976        &self,
977        types: &[WasmValType],
978        builder: &mut FunctionBuilder,
979        values_vec_ptr: Value,
980        values_vec_capacity: Value,
981    ) -> Vec<ir::Value> {
982        let isa = &*self.isa;
983        let value_size = mem::size_of::<u128>();
984
985        debug_assert_enough_capacity_for_length(builder, types.len(), values_vec_capacity);
986
987        // Note that this is little-endian like `store_values_to_array` above,
988        // see notes there for more information.
989        let flags = MemFlags::new()
990            .with_notrap()
991            .with_endianness(ir::Endianness::Little);
992
993        let mut results = Vec::new();
994        for (i, ty) in types.iter().enumerate() {
995            results.push(crate::unbarriered_load_type_at_offset(
996                isa,
997                &mut builder.cursor(),
998                *ty,
999                flags,
1000                values_vec_ptr,
1001                i32::try_from(i * value_size).unwrap(),
1002            ));
1003        }
1004        results
1005    }

1030    /// Invokes the `raise` libcall in `vmctx` if the `succeeded` value
1031    /// indicates if a trap happened.
1032    ///
1033    /// This helper is used when the host returns back to WebAssembly. The host
1034    /// returns a `bool` indicating whether the call succeeded. If the call
1035    /// failed then Cranelift needs to unwind back to the original invocation
1036    /// point. The unwind right now is then implemented in Wasmtime with a
1037    /// `longjmp`, but one day this might be implemented differently with an
1038    /// unwind inside of Cranelift.
1039    ///
1040    /// Additionally in the future for pulley this will emit a special trap
1041    /// opcode for Pulley itself to cease interpretation and exit the
1042    /// interpreter.
1043    pub fn raise_if_host_trapped(
1044        &self,
1045        builder: &mut FunctionBuilder<'_>,
1046        vmctx: ir::Value,
1047        succeeded: ir::Value,
1048    ) {
1049        let trapped_block = builder.create_block();
1050        let continuation_block = builder.create_block();
1051        builder.set_cold_block(trapped_block);
1052        builder
1053            .ins()
1054            .brif(succeeded, continuation_block, &[], trapped_block, &[]);
1055
1056        builder.seal_block(trapped_block);
1057        builder.seal_block(continuation_block);
1058
1059        builder.switch_to_block(trapped_block);
1060        let sigs = BuiltinFunctionSignatures::new(self);
1061        let sig = sigs.host_signature(BuiltinFunctionIndex::raise());
1062        self.call_builtin(builder, vmctx, &[vmctx], BuiltinFunctionIndex::raise(), sig);
1063        builder.ins().trap(TRAP_INTERNAL_ASSERT);
1064
1065        builder.switch_to_block(continuation_block);
1066    }

1068    /// Helper to load the core `builtin` from `vmctx` and invoke it with
1069    /// `args`.
1070    fn call_builtin(
1071        &self,
1072        builder: &mut FunctionBuilder<'_>,
1073        vmctx: ir::Value,
1074        args: &[ir::Value],
1075        builtin: BuiltinFunctionIndex,
1076        sig: ir::Signature,
1077    ) -> ir::Inst {
1078        let isa = &*self.isa;
1079        let ptr_size = isa.pointer_bytes();
1080        let pointer_type = isa.pointer_type();
1081
1082        // Builtins are stored in an array in all `VMContext`s. First load the
1083        // base pointer of the array and then load the entry of the array that
1084        // corresponds to this builtin.
1085        let mem_flags = ir::MemFlags::trusted().with_readonly();
1086        let array_addr = builder.ins().load(
1087            pointer_type,
1088            mem_flags,
1089            vmctx,
1090            i32::from(ptr_size.vmcontext_builtin_functions()),
1091        );
1092        let body_offset = i32::try_from(builtin.index() * pointer_type.bytes()).unwrap();
1093        let func_addr = builder
1094            .ins()
1095            .load(pointer_type, mem_flags, array_addr, body_offset);
1096
1097        let sig = builder.func.import_signature(sig);
1098        self.call_indirect_host(builder, builtin, sig, func_addr, args)
1099    }

1115impl FunctionCompiler<'_> {
1116    fn builder(&mut self, func: ir::Function) -> (FunctionBuilder<'_>, ir::Block) {
1117        self.cx.codegen_context.func = func;
1118        let mut builder = FunctionBuilder::new(
1119            &mut self.cx.codegen_context.func,
1120            self.cx.func_translator.context(),
1121        );
1122
1123        let block0 = builder.create_block();
1124        builder.append_block_params_for_function_params(block0);
1125        builder.switch_to_block(block0);
1126        builder.seal_block(block0);
1127        (builder, block0)
1128    }

1130    fn finish(self, symbol: &str) -> Result<CompiledFunction, CompileError> {
1131        self.finish_with_info(None, symbol)
1132    }
1133
1134    fn finish_with_info(
1135        mut self,
1136        body_and_tunables: Option<(&FunctionBody<'_>, &Tunables)>,
1137        symbol: &str,
1138    ) -> Result<CompiledFunction, CompileError> {
1139        let context = &mut self.cx.codegen_context;
1140        let isa = &*self.compiler.isa;
1141
1142        // Run compilation, but don't propagate the error just yet. This'll
1143        // mutate `context` and the IR contained within (optionally) but it may
1144        // fail if the backend has a bug in it. Use `context` after this
1145        // finishes to optionally emit CLIF and then after that's done actually
1146        // propagate the error if one happened.
1147        let compilation_result =
1148            compile_maybe_cached(context, isa, self.cx.incremental_cache_ctx.as_mut());
1149
1150        if let Some(path) = &self.compiler.clif_dir {
1151            use std::io::Write;
1152
1153            let mut path = path.join(symbol.replace(":", "-"));
1154            path.set_extension("clif");
1155
1156            let mut output = std::fs::File::create(path).unwrap();
1157            write!(
1158                output,
1159                ";; Intermediate Representation of function <{symbol}>:\n",
1160            )
1161            .unwrap();
1162            write!(output, "{}", context.func.display()).unwrap();
1163        }
1164
1165        let compiled_code = compilation_result?;
1166
1167        // Give wasm functions, user defined code, a "preferred" alignment
1168        // instead of the minimum alignment as this can help perf in niche
1169        // situations.
1170        let preferred_alignment = if body_and_tunables.is_some() {
1171            self.compiler.isa.function_alignment().preferred
1172        } else {
1173            1
1174        };
1175
1176        let alignment = compiled_code.buffer.alignment.max(preferred_alignment);
1177        let mut compiled_function = CompiledFunction::new(
1178            compiled_code.buffer.clone(),
1179            context.func.params.user_named_funcs().clone(),
1180            alignment,
1181        );
1182
1183        if let Some((body, tunables)) = body_and_tunables {
1184            let data = body.get_binary_reader();
1185            let offset = data.original_position();
1186            let len = data.bytes_remaining();
1187            compiled_function.set_address_map(
1188                offset.try_into().unwrap(),
1189                len.try_into().unwrap(),
1190                tunables.generate_address_map,
1191            );
1192        }
1193
1194        if isa.flags().unwind_info() {
1195            let unwind = compiled_code
1196                .create_unwind_info(isa)
1197                .map_err(|error| CompileError::Codegen(pretty_error(&context.func, error)))?;
1198
1199            if let Some(unwind_info) = unwind {
1200                compiled_function.set_unwind_info(unwind_info);
1201            }
1202        }
1203
1204        if body_and_tunables
1205            .map(|(_, t)| t.generate_native_debuginfo)
1206            .unwrap_or(false)
1207        {
1208            compiled_function.set_value_labels_ranges(compiled_code.value_labels_ranges.clone());
1209
1210            // DWARF debugging needs the CFA-based unwind information even on Windows.
1211            if !matches!(
1212                compiled_function.metadata().unwind_info,
1213                Some(UnwindInfo::SystemV(_))
1214            ) {
1215                let cfa_unwind = compiled_code
1216                    .create_unwind_info_of_kind(isa, UnwindInfoKind::SystemV)
1217                    .map_err(|error| CompileError::Codegen(pretty_error(&context.func, error)))?;
1218
1219                if let Some(UnwindInfo::SystemV(cfa_unwind_info)) = cfa_unwind {
1220                    compiled_function.set_cfa_unwind_info(cfa_unwind_info);
1221                }
1222            }
1223        }
1224
1225        compiled_function
1226            .set_sized_stack_slots(std::mem::take(&mut context.func.sized_stack_slots));
1227        self.compiler.contexts.lock().unwrap().push(self.cx);
1228
1229        Ok(compiled_function)
1230    }
1231}

1238/// listing of stack maps from Cranelift.
1239fn clif_to_env_stack_maps(
1240    section: &mut StackMapSection,
1241    range: Range<u64>,
1242    clif_stack_maps: &[(CodeOffset, u32, ir::UserStackMap)],
1243) {
1244    for (offset, frame_size, stack_map) in clif_stack_maps {
1245        let mut frame_offsets = Vec::new();
1246        for (ty, frame_offset) in stack_map.entries() {
1247            assert_eq!(ty, ir::types::I32);
1248            frame_offsets.push(frame_offset);
1249        }
1250        let code_offset = range.start + u64::from(*offset);
1251        assert!(code_offset < range.end);
1252        section.push(code_offset, *frame_size, frame_offsets.into_iter());
1253    }
1254}

1256fn declare_and_call(
1257    builder: &mut FunctionBuilder,
1258    signature: ir::Signature,
1259    func_index: u32,
1260    args: &[ir::Value],
1261) -> ir::Inst {
1262    let name = ir::ExternalName::User(builder.func.declare_imported_user_function(
1263        ir::UserExternalName {
1264            namespace: crate::NS_WASM_FUNC,
1265            index: func_index,
1266        },
1267    ));
1268    let signature = builder.func.import_signature(signature);
1269    let callee = builder.func.dfg.ext_funcs.push(ir::ExtFuncData {
1270        name,
1271        signature,
1272        colocated: true,
1273    });
1274    builder.ins().call(callee, &args)
1275}

1314fn save_last_wasm_entry_fp(
1315    builder: &mut FunctionBuilder,
1316    pointer_type: ir::Type,
1317    ptr_size: &impl PtrSize,
1318    vm_store_context_offset: u32,
1319    vmctx: Value,
1320) {
1321    // First we need to get the `VMStoreContext`.
1322    let vm_store_context = builder.ins().load(
1323        pointer_type,
1324        MemFlags::trusted(),
1325        vmctx,
1326        i32::try_from(vm_store_context_offset).unwrap(),
1327    );
1328
1329    // Then store our current stack pointer into the appropriate slot.
1330    let fp = builder.ins().get_frame_pointer(pointer_type);
1331    builder.ins().store(
1332        MemFlags::trusted(),
1333        fp,
1334        vm_store_context,
1335        ptr_size.vmstore_context_last_wasm_entry_fp(),
1336    );
1337}

1339fn save_last_wasm_exit_fp_and_pc(
1340    builder: &mut FunctionBuilder,
1341    pointer_type: ir::Type,
1342    ptr: &impl PtrSize,
1343    limits: Value,
1344) {
1345    // Save the exit Wasm FP to the limits. We dereference the current FP to get
1346    // the previous FP because the current FP is the trampoline's FP, and we
1347    // want the Wasm function's FP, which is the caller of this trampoline.
1348    let trampoline_fp = builder.ins().get_frame_pointer(pointer_type);
1349    let wasm_fp = builder.ins().load(
1350        pointer_type,
1351        MemFlags::trusted(),
1352        trampoline_fp,
1353        // The FP always points to the next older FP for all supported
1354        // targets. See assertion in
1355        // `crates/wasmtime/src/runtime/vm/traphandlers/backtrace.rs`.
1356        0,
1357    );
1358    builder.ins().store(
1359        MemFlags::trusted(),
1360        wasm_fp,
1361        limits,
1362        ptr.vmstore_context_last_wasm_exit_fp(),
1363    );
1364    // Finally save the Wasm return address to the limits.
1365    let wasm_pc = builder.ins().get_return_address(pointer_type);
1366    builder.ins().store(
1367        MemFlags::trusted(),
1368        wasm_pc,
1369        limits,
1370        ptr.vmstore_context_last_wasm_exit_pc(),
1371    );
